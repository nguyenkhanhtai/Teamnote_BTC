\Algorithm{Công thức số học}
{Số học modulo, rút gọn số mũ Euler mở rộng, Legendre, Wilson, Frobenius Coin, Floor Trick, CRT, Fibonacci}
{}
{}{}

\begin{itemize}
    \item \textbf{Rút gọn số mũ Euler mở rộng} ($\forall a, m$):
    $$ a^b \equiv a^{(b \bmod \phi(m)) + \phi(m)} \pmod m \quad (b \ge \phi(m)) $$
    \item \textbf{Công thức Legendre ($v_p(n!)$)}: $v_p(n!) = \sum_{k=1}^\infty \lfloor \frac{n}{p^k} \rfloor = \frac{n - S_p(n)}{p - 1}$.
    \item \textbf{LTE}: $v_p(x)$ là số mũ của nguyên tố $p$ trong $x\ne0$. Với $a,b,n$ nguyên dương, $a\ne b$, $n\ge1$: nếu $p$ lẻ, $p\mid a-b$, $p\nmid ab$, thì $v_p(a^n-b^n)=v_p(a-b)+v_p(n)$. Nếu $p$ lẻ, $p\mid a+b$, $p\nmid ab$, $n$ lẻ: $v_p(a^n+b^n)=v_p(a+b)+v_p(n)$. Với $a,b$ lẻ và $p=2$: $n$ lẻ cho $v_2(a^n-b^n)=v_2(a-b)$; $n$ chẵn cho $v_2(a^n-b^n)=v_2(a-b)+v_2(a+b)+v_2(n)-1$. Dùng tính mức chia hết của hiệu/tổng lũy thừa mà không tính số lớn.
    \item \textbf{Định lý Wilson}: $(p - 1)! \equiv -1 \pmod p \iff p$ nguyên tố $\implies (p - 2)! \equiv 1 \pmod p$.
    \item \textbf{Frobenius Coin (Chicken McNugget)}: Với $\gcd(a,b)=1$, số lớn nhất không biểu diễn được dạng $ax+by$ ($x,y\ge0$) là $ab-a-b$; số lượng không biểu diễn được là $\frac{(a-1)(b-1)}{2}$.
    \item \textbf{Floor Trick ($O(\sqrt{n})$ nhóm)}:
\begin{lstlisting}
for (int l = 1, r; l <= n; l = r + 1) {
    int v = n / l; r = n / v;
}
\end{lstlisting}
    \item \textbf{Định lý số dư Trung Hoa (CRT)}: $x \equiv a_i \pmod{m_i}$.
    $$ x \equiv \sum_{i=1}^k a_i M_i (M_i^{-1} \bmod m_i) \pmod M \quad (M = \prod m_i, M_i = M/m_i) $$
    \item \textbf{Fibonacci}: $\gcd(F_m, F_n) = F_{\gcd(m, n)}$, $F_{2n} = F_n(2F_{n+1} - F_n)$.
\end{itemize}

\noindent\textbf{Bézout (tổng quát)}: Với $a_1,\ldots,a_n\in\mathbb Z$ không đồng thời bằng 0, $g=\gcd(|a_1|,\ldots,|a_n|)$: tồn tại $c_i\in\mathbb Z$ sao cho $\sum_i c_i a_i=g$. Mọi tổ hợp nguyên tạo đúng các bội của $g$: $\sum_i c_i a_i=x$ có nghiệm iff $g\mid x$. \textbf{Cue}: cộng/trừ các bước tùy ý $\to$ gcd các bước chia hết độ dịch chuyển; nghịch đảo modulo $\to ax+my=1$ (cần $\gcd(a,m)=1$). Hệ số được âm; yêu cầu $c_i\ge0$ thì gcd chia hết chưa đủ. Extended GCD tìm hệ số cho hai số, ghép dần cho nhiều số.

\noindent\textbf{GCD: cộng/trừ và đổi mốc}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parsep}{0pt}
    \item $\gcd(u,v)=\gcd(u,v+ku)=\gcd(v,u\bmod v)$ ($k\in\mathbb Z$, $v\ne0$ cho modulo); $\gcd(cu,cv)=|c|\gcd(u,v)$. Cộng/trừ bội giữ các ước chung.
    \item Đặt $d_i=a_{i+1}-a_i$, $g=\gcd|d_i|$: $\gcd(a_1,\ldots,a_n)=\gcd(a_1,g)$; tịnh tiến: $\gcd(a_1+b,\ldots,a_n+b)=\gcd(a_1+b,g)$ (Row GCD). Phải giữ mốc; $\gcd(z,0)=|z|$.
    \item $\gcd(pu+qv,ru+sv)=\gcd(u,v)$ nếu $ps-qr=\pm1$ (phép ngược có hệ số nguyên). Tổ hợp tùy ý chưa chắc giữ gcd, vd. $(u+v,u-v)$.
\end{itemize}

\Algorithm{Hàm nhân tính và M\"obius Inversion}
{Tích chập Dirichlet, nghịch đảo M\"obius, tổng GCD/LCM}
{}
{}{}

\begin{itemize}
    \item \textbf{Euler Phi \& M\"obius}: $\phi(n) = n \prod_{p \mid n} (1 - 1/p)$, $\sum_{d \mid n} \phi(d) = n$, $\sum_{d \mid n} \mu(d) = [n = 1]$.
    \item \textbf{Dirichlet \& Inversion}: $(f * g)(n) = \sum_{d \mid n} f(d) g(n / d)$.
    $$ f(n) = \sum_{d \mid n} g(d) \iff g(n) = \sum_{d \mid n} \mu(d) f(n / d) $$
    \item \textbf{Đếm cặp nguyên tố cùng nhau}:
    $$ \sum_{i=1}^n \sum_{j=1}^m [\gcd(i, j) = 1] = \sum_{d=1}^{\min(n, m)} \mu(d) \left\lfloor \frac{n}{d} \right\rfloor \left\lfloor \frac{m}{d} \right\rfloor $$
    \item \textbf{Tổng GCD trên mọi cặp}:
    $$ \sum_{i=1}^n \sum_{j=1}^m \gcd(i, j) = \sum_{g=1}^{\min(n, m)} \phi(g) \left\lfloor \frac{n}{g} \right\rfloor \left\lfloor \frac{m}{g} \right\rfloor $$
    \item \textbf{Tổng LCM trên mọi cặp} ($S(x) = \frac{x(x+1)}{2}$):
    $$ \sum_{i \le n, j \le m} \text{lcm}(i, j) = \sum_{g} g \sum_{d} d^2 \mu(d) S\left(\left\lfloor \frac{n}{gd} \right\rfloor\right) S\left(\left\lfloor \frac{m}{gd} \right\rfloor\right) $$
\end{itemize}

\Algorithm{Công thức tổ hợp}
{Đẳng thức nhị thức, Hockey-stick, Vandermonde, Inversion, Stars and Bars, Catalan, Stirling, Cayley, LGV}
{}
{}{}

\begin{itemize}
    \item \textbf{Đẳng thức nhị thức}: $\binom{n}{k} = \binom{n}{n-k} = \binom{n-1}{k-1} + \binom{n-1}{k}$; $k \binom{n}{k} = n \binom{n-1}{k-1}$; $\frac{1}{k+1}\binom{n}{k} = \frac{1}{n+1}\binom{n+1}{k+1}$; $\binom{n}{k} \binom{k}{m} = \binom{n}{m} \binom{n-m}{k-m}$; Đối số trên âm: $\binom{-n}{k} = (-1)^k \binom{n+k-1}{k}$.
    \item \textbf{Tổng nhị thức}: $\sum_{k=0}^n \binom{n}{k} = 2^n$; $\sum_{k=0}^n (-1)^k \binom{n}{k} = [n = 0]$; $\sum_{k \text{ chẵn}} \binom{n}{k} = \sum_{k \text{ lẻ}} \binom{n}{k} = 2^{n-1}$; $\sum_{k=0}^n k \binom{n}{k} = n 2^{n-1}$; $\sum_{k=0}^n k^2 \binom{n}{k} = n(n+1) 2^{n-2}$; $\sum_{k=0}^n \frac{1}{k+1}\binom{n}{k} = \frac{2^{n+1}-1}{n+1}$.
    \item \textbf{Hockey-Stick \& Vandermonde}: $\sum_{i=r}^n \binom{i}{r} = \binom{n+1}{r+1}$; $\sum_{i=0}^k \binom{n+i}{i} = \binom{n+k+1}{k}$; $\sum_{k=0}^r \binom{m}{k} \binom{n}{r-k} = \binom{m+n}{r} \implies \sum_{k=0}^n \binom{n}{k}^2 = \binom{2n}{n}$.
    \item \textbf{Binomial Inversion và sai phân}: $f(n) = \sum_{k=0}^n \binom{n}{k} g(k) \iff g(n) = \sum_{k=0}^n (-1)^{n-k} \binom{n}{k} f(k)$; Dạng tổng phía trên: $f(n) = \sum_{k=n}^N \binom{k}{n} g(k) \iff g(n) = \sum_{k=n}^N (-1)^{k-n} \binom{k}{n} f(k)$; $\sum_{k=0}^n (-1)^{n-k} \binom{n}{k} k^m = n! \left\{ \begin{matrix} m \\ n \end{matrix} \right\}$ ($0$ nếu $m<n$, $n!$ nếu $m=n$).
    \item \textbf{Stars and Bars}: $x_i \ge 0 \implies \binom{n + k - 1}{k - 1}$; $x_i \ge 1 \implies \binom{n - 1}{k - 1}$; $x_i \ge c_i \implies n' = n - \sum c_i$.
    \item \textbf{Derangements}: $D_0 = 1, D_1 = 0, D_n = (n - 1)(D_{n-1} + D_{n-2}) = n D_{n-1} + (-1)^n \approx [n! / e]$.
    \item \textbf{Số Catalan $C_n = \frac{1}{n+1}\binom{2n}{n}$}: Dãy ngoặc đúng, cây nhị phân đầy đủ có $n+1$ lá, chia tam giác đa giác lồi $(n+2)$ đỉnh, đường Dyck không vượt $y=x$, hoán vị sinh bằng một stack.
    \item \textbf{Nguyên lý phản xạ (Reflection Principle)}: Đường đi $(0,0)\to(n,m)$ không chạm $y = x + k$ ($k > 0$): $\binom{n + m}{m} - \binom{n + m}{m - k}$.
    \item \textbf{Định lý Ballot}: A có $a$ phiếu, B có $b$ ($a>b$). Xác suất A luôn dẫn trước nghiêm ngặt: $\frac{a - b}{a + b}$.
    \item \textbf{Số Stirling loại 1 và 2}: $\left[ \begin{matrix} n \\ k \end{matrix} \right] = \left[ \begin{matrix} n-1 \\ k-1 \end{matrix} \right] + (n-1) \left[ \begin{matrix} n-1 \\ k \end{matrix} \right]$, $\left\{ \begin{matrix} n \\ k \end{matrix} \right\} = \left\{ \begin{matrix} n-1 \\ k-1 \end{matrix} \right\} + k \left\{ \begin{matrix} n-1 \\ k \end{matrix} \right\}$.
    $x^n = \sum_{k=0}^n \left\{ \begin{matrix} n \\ k \end{matrix} \right\} x^{\underline{k}} \implies \sum_{i=1}^n i^k = \sum_{j=0}^k \left\{ \begin{matrix} k \\ j \end{matrix} \right\} j! \binom{n + 1}{j + 1}$.
    \item \textbf{Chu kỳ hoán vị}: Với hoán vị $\pi$ của $n$ phần tử có $c$ chu kỳ (tính cả điểm cố định), số phép đổi chỗ hai vị trí bất kỳ ít nhất để về identity là $n-c$. Mỗi chu kỳ dài $\ell$ cần $\ell-1$ lần. Biến hoán vị $A$ thành $B$: xét $\pi(i)=\operatorname{pos}_B(A_i)$. Không áp dụng cho chỉ đổi hai vị trí kề nhau.
    \item \textbf{Eulerian numbers}: $E(n,k)$ đếm hoán vị $n$ phần tử có đúng $k$ vị trí giảm $a_i>a_{i+1}$. $E(0,0)=1$; với $n\ge1$, $E(n,k)=(n-k)E(n-1,k-1)+(k+1)E(n-1,k)$, chỉ tính $0\le k<n$, các chỉ số khác bằng $0$. Với $n\ge1$, cũng bằng số hoán vị có đúng $k$ vị trí $a_i>i$ (chỉ số từ $1$).
    \item \textbf{PIE}: $|\bigcup_{i=1}^n A_i| = \sum_{\emptyset \ne J} (-1)^{|J|-1} |\bigcap_{j \in J} A_j|$; Thuộc đúng $r$ tập: $\sum_{m=r}^n (-1)^{m-r} \binom{m}{r} \sum_{|X|=m} |\bigcap_{i \in X} A_i|$.
    \item \textbf{Định lý Lucas}: $\binom{n}{k} \equiv \prod_{i} \binom{n_i}{k_i} \pmod p$ (chữ số cơ số $p$).
    \item \textbf{Đếm cây và Cayley}: $n^{n-2}$ cây có nhãn, rừng gồm $k$ cây có gốc: $k n^{n-k-1}$.
    \item \textbf{LGV (Lindstr\"om--Gessel--Viennot)}: Trên DAG, đặt $M_{ij}$ là số đường đi $A_i\to B_j$. Khi đó $\det M=\sum_{\sigma\in S_k}\operatorname{sgn}(\sigma)N_\sigma$, với $N_\sigma$ đếm các bộ đường đi đôi một không chung đỉnh, nối $A_i\to B_{\sigma(i)}$. Nếu mọi phép ghép khác đồng nhất đều buộc giao nhau, $\det M$ chính là số bộ nối $A_i\to B_i$. Nhận biết: nhiều đường đi đơn điệu trên lưới, không chung đỉnh; mỗi $M_{ij}=\binom{\Delta x+\Delta y}{\Delta x}$ nếu $\Delta x,\Delta y\ge0$, ngược lại $0$. Các bộ giao nhau triệt tiêu bằng cách đổi đuôi sau điểm giao, làm đổi dấu hoán vị.
    \item \textbf{Matrix--Tree (Kirchhoff)}: Đồ thị vô hướng có Laplacian $L_{ii}=\deg(i)$, $L_{ij}=-m_{ij}$ ($i\ne j$), với $m_{ij}$ là số cạnh nối $i,j$. Xóa hàng và cột cùng chỉ số $r$ được $L^{(r)}$; số cây khung $\tau(G)=\det L^{(r)}$, với mọi $r$. Đồ thị không liên thông cho $0$; một đỉnh cho $1$. Giữ cạnh song song, bỏ self-loop. Tính modulo nguyên tố bằng \texttt{determinant\_mod}.
    \item \textbf{Đếm MST}: Kruskal theo nhóm cạnh cùng trọng số; co DSU từ các cạnh nhẹ hơn, bỏ cạnh nội bộ. Nhân $\tau(C)$ cho từng thành phần liên thông $C$ của đồ thị co (giữ cạnh song song), rồi mới union nhóm hiện tại. Đồ thị gốc không liên thông: không có MST.
\end{itemize}

\Algorithm{Lagrange}
{Nội suy Lagrange tại $X$ từ $n+1$ điểm cách đều có giá trị $y_0,\dots,y_n$}
{$O(n)$ tính giá trị}
{compactcpp}{source/cpp/math/lagrange.cpp}
\textbf{Công thức:} Với $d+1$ điểm phân biệt $(x_i,y_i)$ và $\deg f\le d$,
\[
f(X)=\sum_{i=0}^{d}y_i\prod_{\substack{0\le j\le d\\j\ne i}}\frac{X-x_j}{x_i-x_j}.
\]
Code dùng $x_i=i$, nên $\prod_{j\ne i}(i-j)=(-1)^{d-i}i!(d-i)!$:
\[
f(X)=\sum_{i=0}^{d}y_i\frac{(-1)^{d-i}}{i!(d-i)!}
\prod_{\substack{0\le j\le d\\j\ne i}}(X-j).
\]
Trong modulo $p$, phép chia là nhân nghịch đảo; tích tiền tố/hậu tố tính các tích bỏ $i$ trong $O(d)$.\\
\textbf{Khi dùng:} Cần $f(N)$ với $N$ rất lớn, biết $f$ là đa thức bậc tối đa $d$ và tính được $f(0),\dots,f(d)$ nhanh. Không suy ra tính đa thức chỉ từ vài giá trị đầu.\\
\textbf{Ví dụ:} $S_k(N)=\sum_{i=1}^{N}i^k$ có bậc $k+1$: lập $y[0..k+1]$ bằng tổng tiền tố, gọi \texttt{lagrange(y,N,p)}. Tương tự, đếm cách chọn $k$ phần tử từ $N$ phần tử: $\binom{N}{k}$ là đa thức bậc $k$.\\
\textbf{Điều kiện code:} $y[i]=f(i)$, modulo nguyên tố $p$, số điểm $d+1<p$; với tổng lũy thừa cần $k+2<p$. Mỗi lần tính tốn $O(d+\log p)$, gồm nghịch đảo modulo.

\Algorithm{Công thức Formal Power Series (FPS)}
{Công thức thao tác đa thức bằng phương pháp Newton và convolution}
{}
{}{}

\begin{itemize}
    \item \textbf{Nghịch đảo ($Q=P^{-1}$)}: $Q_k \equiv Q_{k-1}(2 - P Q_{k-1}) \pmod{x^{2^k}}$, khởi đầu $Q_0 = P(0)^{-1}$.
    \item \textbf{Đạo hàm và tích phân}: $P'(x) = \sum i a_i x^{i-1}$, $\int P(x) = \sum \frac{a_i}{i+1} x^{i+1}$.
    \item \textbf{Logarithm}: $\ln P(x) = \int \frac{P'(x)}{P(x)} dx \pmod{x^n}$ (cần $P(0) = 1$).
    \item \textbf{Hàm mũ}: $Q_k \equiv Q_{k-1}(1 - \ln Q_{k-1} + P) \pmod{x^{2^k}}$ (cần $P(0) = 0$).
    \item \textbf{Căn bậc hai}: $Q_k \equiv \frac{1}{2}(Q_{k-1} + P Q_{k-1}^{-1}) \pmod{x^{2^k}}$ ($P(0)$ là thặng dư bậc hai).
    \item \textbf{Lũy thừa}: $P(x)^k = \exp(k \ln P(x))$. Nếu $P(x) = c x^d (1 + Q(x))$, $P(x)^k = c^k x^{k d} \exp(k \ln(1 + Q(x)))$.
    \item \textbf{Chia lấy thương và dư ($A = BQ + R$)}: Với $\text{rev}_k(P) = x^k P(1/x)$:
    $$\text{rev}_{n-m}(Q) \equiv \text{rev}_n(A) \cdot (\text{rev}_m(B))^{-1} \pmod{x^{n-m+1}}, \quad R = A - BQ$$
    \item \textbf{Taylor Shift $P(x + c)$}: $P(x + c) = \sum_{k=0}^n \frac{x^k}{k!} \sum_{i=k}^n (i! a_i) \frac{c^{i-k}}{(i-k)!}$. Convolution dãy $A_i=i!a_i$ đảo ngược với $B_j = \frac{c^j}{j!}$.
    \item \textbf{Convolution modulo tùy ý (MTT)}: Tách $A=A_1B+A_0$, $B= \lceil\sqrt{M}\rceil \approx 32000$. Tính $(A_1B_1),(A_1B_0+A_0B_1),(A_0B_0)$ bằng bốn FFT.
\end{itemize}

\subsection{Game: Nim và chiến lược}
% References: MIT ES.268 Theory of Impartial Games; CMU Combinatorics F11/ex4a.pdf.
\begin{itemize}
    \item \textbf{Nim}: mỗi lượt giảm một đống; người lấy cuối thắng. Thua $\iff X=\bigoplus_i a_i=0$. Nếu $X\ne0$, chọn đống có bit cao nhất của $X$, đổi $a_i\to a_i\oplus X<a_i$ để trả XOR về $0$.
    \item \textbf{Mis\`ere Nim} (lấy cuối thua): mọi đống $\le1$ thì thua khi số đống $1$ lẻ; còn đống $>1$ thì thua $\iff X=0$. \emph{Cách đi}: còn ít nhất hai đống lớn thì chơi như Nim; chỉ còn một đống lớn thì giảm nó về $0$ hoặc $1$ để để lại số đống $1$ lẻ.
    \item \textbf{Staircase Nim}: chuyển tùy số viên từ bậc $i$ xuống $i-1$; bậc $0$ không chơi tiếp. Chỉ XOR các bậc \emph{lẻ}. Đối thủ chuyển từ bậc chẵn xuống lẻ thì chuyển đúng số viên ấy xuống thêm một bậc, khôi phục XOR.
    \item \textbf{Wythoff Nim}: hai đống, giảm một đống hoặc giảm cả hai cùng số lượng. Với $a\le b$, đặt $k=b-a$: thua $\iff a=\lfloor k\varphi\rfloor$, $\varphi=(1+\sqrt5)/2$. Ví dụ $(1,2),(3,5),(4,7)$; số lớn cần tính floor chính xác.
    \item \textbf{Green Hackenbush trên cây}: chặt cạnh, phần mất nối với gốc biến mất. Gộp nhánh bằng XOR; thêm một cạnh dưới nhánh thì SG tăng $1$: $g(u)=\bigoplus_{v\text{ con}}(g(v)+1)$. Chỉ dùng cho normal play với mọi cạnh đều được cả hai người chặt.
    \item \textbf{Pairing / mirror} (hết nước đi thì thua): ghép các nước thành cặp; đáp bằng nước còn lại để khôi phục đối xứng. Phải chứng minh nước đáp \emph{luôn còn hợp lệ}, không chỉ bàn ban đầu đối xứng. Nếu đi tâm tạo được cấu hình này thì người đầu có thể chiếm chiến lược phản chiếu.
    \item \textbf{Strategy stealing}: game đặt quân hữu hạn, không hòa; nếu một nước thêm không bao giờ gây bất lợi, giả sử người sau có chiến lược thắng rồi người trước đi thêm và ``mượn'' chiến lược đó để tạo mâu thuẫn. Chứng minh người đầu thắng nhưng thường không chỉ ra nước thắng cụ thể.
\end{itemize}

\Algorithm{Xác suất và kỳ vọng}
{Tính tuyến tính của kỳ vọng, biến chỉ báo, Coupon Collector, bất đẳng thức}
{}
{}{}

\begin{itemize}
    \item \textbf{Tính tuyến tính của kỳ vọng}: $E[X_1 + \dots + X_n] = E[X_1] + \dots + E[X_n]$ ($\forall X_i$).
    \item \textbf{Biến chỉ báo}: $E[X] = \sum P(A_i)$.
    \item \textbf{Phân phối hình học}: $E[X] = 1/p$, $\text{Var}(X) = (1-p)/p^2$.
    \item \textbf{Coupon Collector}: $E[T] = n \sum_{i=1}^n \frac{1}{i} = n H_n \approx n(\ln n + \gamma)$ ($\gamma \approx 0.5772$).
    \item \textbf{Markov}: $P(X \ge a) \le \frac{E[X]}{a}$; \textbf{Chebyshev}: $P(|X - \mu| \ge k\sigma) \le \frac{1}{k^2}$.
\end{itemize}
