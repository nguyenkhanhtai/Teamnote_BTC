\Algorithm{Number Theory Formulas}
{Modular arithmetic, Extended Euler Power Reduction, Legendre, Wilson, Frobenius Coin, Floor Trick, CRT, Fibonacci}
{}
{}{}

\begin{itemize}
    \item \textbf{Extended Euler Totient Power Reduction} ($\forall a, m$):
    $$ a^b \equiv a^{(b \bmod \phi(m)) + \phi(m)} \pmod m \quad (b \ge \phi(m)) $$
    \item \textbf{Legendre's Formula ($v_p(n!)$)}: $v_p(n!) = \sum_{k=1}^\infty \lfloor \frac{n}{p^k} \rfloor = \frac{n - S_p(n)}{p - 1}$.
    \item \textbf{Wilson's Theorem}: $(p - 1)! \equiv -1 \pmod p \iff p$ prime $\implies (p - 2)! \equiv 1 \pmod p$.
    \item \textbf{Frobenius Coin (Chicken McNugget)}: For $\gcd(a, b) = 1$, max non-representable $ax + by$ ($x, y \ge 0$) is $ab - a - b$; total non-representable count is $\frac{(a-1)(b-1)}{2}$.
    \item \textbf{Floor Trick ($O(\sqrt{n})$ Blocks)}:
\begin{lstlisting}
for (int l = 1, r; l <= n; l = r + 1) {
    int v = n / l; r = n / v;
}
\end{lstlisting}
    \item \textbf{Chinese Remainder Theorem (CRT)}: $x \equiv a_i \pmod{m_i}$.
    $$ x \equiv \sum_{i=1}^k a_i M_i (M_i^{-1} \bmod m_i) \pmod M \quad (M = \prod m_i, M_i = M/m_i) $$
    \item \textbf{Fibonacci}: $\gcd(F_m, F_n) = F_{\gcd(m, n)}$, $F_{2n} = F_n(2F_{n+1} - F_n)$.
\end{itemize}

\Algorithm{Multiplicative Functions \& M\"obius Inversion}
{Dirichlet Convolution, M\"obius Inversion, GCD/LCM summations}
{}
{}{}

\begin{itemize}
    \item \textbf{Euler Phi \& M\"obius}: $\phi(n) = n \prod_{p \mid n} (1 - 1/p)$, $\sum_{d \mid n} \phi(d) = n$, $\sum_{d \mid n} \mu(d) = [n = 1]$.
    \item \textbf{Dirichlet \& Inversion}: $(f * g)(n) = \sum_{d \mid n} f(d) g(n / d)$.
    $$ f(n) = \sum_{d \mid n} g(d) \iff g(n) = \sum_{d \mid n} \mu(d) f(n / d) $$
    \item \textbf{Count coprime pairs}:
    $$ \sum_{i=1}^n \sum_{j=1}^m [\gcd(i, j) = 1] = \sum_{d=1}^{\min(n, m)} \mu(d) \left\lfloor \frac{n}{d} \right\rfloor \left\lfloor \frac{m}{d} \right\rfloor $$
    \item \textbf{Sum of GCD over all pairs}:
    $$ \sum_{i=1}^n \sum_{j=1}^m \gcd(i, j) = \sum_{g=1}^{\min(n, m)} \phi(g) \left\lfloor \frac{n}{g} \right\rfloor \left\lfloor \frac{m}{g} \right\rfloor $$
    \item \textbf{Sum of LCM over all pairs} ($S(x) = \frac{x(x+1)}{2}$):
    $$ \sum_{i \le n, j \le m} \text{lcm}(i, j) = \sum_{g} g \sum_{d} d^2 \mu(d) S\left(\left\lfloor \frac{n}{gd} \right\rfloor\right) S\left(\left\lfloor \frac{m}{gd} \right\rfloor\right) $$
\end{itemize}

\Algorithm{Combinatorics Formulas}
{Binomial identities, Hockey-stick, Vandermonde, Inversion, Stars and bars, Catalan, Stirling, Cayley, LGV}
{}
{}{}

\begin{itemize}
    \item \textbf{Binomial Identities}: $\binom{n}{k} = \binom{n}{n-k} = \binom{n-1}{k-1} + \binom{n-1}{k}$; $k \binom{n}{k} = n \binom{n-1}{k-1}$; $\frac{1}{k+1}\binom{n}{k} = \frac{1}{n+1}\binom{n+1}{k+1}$; $\binom{n}{k} \binom{k}{m} = \binom{n}{m} \binom{n-m}{k-m}$; Upper negation: $\binom{-n}{k} = (-1)^k \binom{n+k-1}{k}$.
    \item \textbf{Binomial Summations}: $\sum_{k=0}^n \binom{n}{k} = 2^n$; $\sum_{k=0}^n (-1)^k \binom{n}{k} = [n = 0]$; $\sum_{k \text{ even}} \binom{n}{k} = \sum_{k \text{ odd}} \binom{n}{k} = 2^{n-1}$; $\sum_{k=0}^n k \binom{n}{k} = n 2^{n-1}$; $\sum_{k=0}^n k^2 \binom{n}{k} = n(n+1) 2^{n-2}$; $\sum_{k=0}^n \frac{1}{k+1}\binom{n}{k} = \frac{2^{n+1}-1}{n+1}$.
    \item \textbf{Hockey-Stick \& Vandermonde}: $\sum_{i=r}^n \binom{i}{r} = \binom{n+1}{r+1}$; $\sum_{i=0}^k \binom{n+i}{i} = \binom{n+k+1}{k}$; $\sum_{k=0}^r \binom{m}{k} \binom{n}{r-k} = \binom{m+n}{r} \implies \sum_{k=0}^n \binom{n}{k}^2 = \binom{2n}{n}$.
    \item \textbf{Binomial Inversion \& Differences}: $f(n) = \sum_{k=0}^n \binom{n}{k} g(k) \iff g(n) = \sum_{k=0}^n (-1)^{n-k} \binom{n}{k} f(k)$; Upper: $f(n) = \sum_{k=n}^N \binom{k}{n} g(k) \iff g(n) = \sum_{k=n}^N (-1)^{k-n} \binom{k}{n} f(k)$; $\sum_{k=0}^n (-1)^{n-k} \binom{n}{k} k^m = n! \left\{ \begin{matrix} m \\ n \end{matrix} \right\}$ ($0$ if $m < n$, $n!$ if $m=n$).
    \item \textbf{Stars and Bars}: $x_i \ge 0 \implies \binom{n + k - 1}{k - 1}$; $x_i \ge 1 \implies \binom{n - 1}{k - 1}$; $x_i \ge c_i \implies n' = n - \sum c_i$.
    \item \textbf{Derangements}: $D_0 = 1, D_1 = 0, D_n = (n - 1)(D_{n-1} + D_{n-2}) = n D_{n-1} + (-1)^n \approx [n! / e]$.
    \item \textbf{Catalan Numbers $C_n = \frac{1}{n+1}\binom{2n}{n}$}: Valid bracketings, Full binary trees with $n+1$ leaves, Triangulations of convex $(n+2)$-gon, Dyck paths not crossing $y=x$, 1-Stack permutations.
    \item \textbf{Reflection Principle}: Paths $(0, 0) \to (n, m)$ not touching $y = x + k$ ($k > 0$): $\binom{n + m}{m} - \binom{n + m}{m - k}$.
    \item \textbf{Balloting Theorem}: A has $a$ votes, B has $b$ ($a > b$). Prob A strictly leads throughout: $\frac{a - b}{a + b}$.
    \item \textbf{Stirling Numbers of 1st \& 2nd Kind}: $\left[ \begin{matrix} n \\ k \end{matrix} \right] = \left[ \begin{matrix} n-1 \\ k-1 \end{matrix} \right] + (n-1) \left[ \begin{matrix} n-1 \\ k \end{matrix} \right]$, $\left\{ \begin{matrix} n \\ k \end{matrix} \right\} = \left\{ \begin{matrix} n-1 \\ k-1 \end{matrix} \right\} + k \left\{ \begin{matrix} n-1 \\ k \end{matrix} \right\}$.
    $x^n = \sum_{k=0}^n \left\{ \begin{matrix} n \\ k \end{matrix} \right\} x^{\underline{k}} \implies \sum_{i=1}^n i^k = \sum_{j=0}^k \left\{ \begin{matrix} k \\ j \end{matrix} \right\} j! \binom{n + 1}{j + 1}$.
    \item \textbf{PIE}: $|\bigcup_{i=1}^n A_i| = \sum_{\emptyset \ne J} (-1)^{|J|-1} |\bigcap_{j \in J} A_j|$; Exactly $r$ sets: $\sum_{m=r}^n (-1)^{m-r} \binom{m}{r} \sum_{|X|=m} |\bigcap_{i \in X} A_i|$.
    \item \textbf{Lucas' Theorem}: $\binom{n}{k} \equiv \prod_{i} \binom{n_i}{k_i} \pmod p$ (base-$p$ digits).
    \item \textbf{Tree Counting \& Cayley}: $n^{n-2}$ labeled trees, Forest of $k$ rooted trees $k n^{n-k-1}$, Matrix Tree Theorem ($\det(L_i)$), LGV Lemma ($\det(M)$).
\end{itemize}

\Algorithm{Formal Power Series (FPS) Formulas}
{Key formulas for polynomial operations via Newton's method \& convolution}
{}
{}{}

\begin{itemize}
    \item \textbf{Inverse ($Q = P^{-1}$)}: $Q_k \equiv Q_{k-1}(2 - P Q_{k-1}) \pmod{x^{2^k}}$, base $Q_0 = P(0)^{-1}$.
    \item \textbf{Derivative \& Integral}: $P'(x) = \sum i a_i x^{i-1}$, $\int P(x) = \sum \frac{a_i}{i+1} x^{i+1}$.
    \item \textbf{Logarithm}: $\ln P(x) = \int \frac{P'(x)}{P(x)} dx \pmod{x^n}$ (requires $P(0) = 1$).
    \item \textbf{Exponential}: $Q_k \equiv Q_{k-1}(1 - \ln Q_{k-1} + P) \pmod{x^{2^k}}$ (requires $P(0) = 0$).
    \item \textbf{Square Root}: $Q_k \equiv \frac{1}{2}(Q_{k-1} + P Q_{k-1}^{-1}) \pmod{x^{2^k}}$ ($P(0)$ is quadratic residue).
    \item \textbf{Power}: $P(x)^k = \exp(k \ln P(x))$. If $P(x) = c x^d (1 + Q(x))$, $P(x)^k = c^k x^{k d} \exp(k \ln(1 + Q(x)))$.
    \item \textbf{Division with Remainder ($A = BQ + R$)}: With $\text{rev}_k(P) = x^k P(1/x)$:
    $$\text{rev}_{n-m}(Q) \equiv \text{rev}_n(A) \cdot (\text{rev}_m(B))^{-1} \pmod{x^{n-m+1}}, \quad R = A - BQ$$
    \item \textbf{Taylor Shift $P(x + c)$}: $P(x + c) = \sum_{k=0}^n \frac{x^k}{k!} \sum_{i=k}^n (i! a_i) \frac{c^{i-k}}{(i-k)!}$. Convolute reversed $A_i = i! a_i$ with $B_j = \frac{c^j}{j!}$.
    \item \textbf{Any-Mod Convolution (MTT)}: Split $A = A_1 B + A_0$, $B = \lceil\sqrt{M}\rceil \approx 32000$. Compute $(A_1 B_1), (A_1 B_0 + A_0 B_1), (A_0 B_0)$ with 4 FFTs.
\end{itemize}

\Algorithm{Burnside Lemma \& P\'olya Theorem}
{Count distinct equivalence classes under group actions, Necklaces, Bracelets}
{}
{}{}

\begin{itemize}
    \item $|Classes| = \frac{1}{|G|} \sum_{g \in G} |Fix(g)| = \frac{1}{|G|} \sum_{g \in G} k^{c(g)}$.
    \item \textbf{Necklace ($n$ rotations only)}: $N(n, k) = \frac{1}{n} \sum_{d \mid n} \phi(d) k^{n / d}$.
    \item \textbf{Bracelet (Rotations + Reflections)}:
    \begin{itemize}
        \item $n$ odd: $B(n, k) = \frac{1}{2} N(n, k) + \frac{1}{2} k^{(n+1)/2}$.
        \item $n$ even: $B(n, k) = \frac{1}{2} N(n, k) + \frac{1}{4} (k^{n/2} + k^{n/2 + 1})$.
    \end{itemize}
\end{itemize}

\Algorithm{Game Strategy \& State Reasoning}
{Winning/Losing state axioms, Retrograde analysis, Symmetry, Parity, Sprague-Grundy MEX, Nim variants}
{$O(V + E)$}
{}{source/pseudocode/game_strategy.txt}

\begin{itemize}
    \item \textbf{Winning ($\mathcal{W}$ / N-pos) vs Losing ($\mathcal{L}$ / P-pos) State Axioms}:
    \begin{itemize}
        \item \textbf{Terminal State}: Normal play (no moves $\implies \mathcal{L}$); Mis\`ere play (no moves $\implies \mathcal{W}$).
        \item \textbf{Winning State $\mathcal{W}$}: $\exists v \in \text{next}(u)$ such that $v \in \mathcal{L}$ (at least ONE move leads to an $\mathcal{L}$-state).
        \item \textbf{Losing State $\mathcal{L}$}: $\forall v \in \text{next}(u)$, $v \in \mathcal{W}$ (ALL moves lead to $\mathcal{W}$-states; or no moves exist).
        \item \textbf{Optimal Play}: From $\mathcal{W}$, always choose a move to $\mathcal{L}$. Opponent from $\mathcal{L}$ is forced to move back to $\mathcal{W}$.
    \end{itemize}
    \item \textbf{Problem-Solving Deduction Workflow}:
    \begin{itemize}
        \item \textbf{Retrograde BFS ($O(V + E)$)}: Propagate outcomes backward from terminal states using out-degree counters; unvisited states in cycles are \textbf{DRAW}.
        \item \textbf{Symmetry / Mirroring}: If P2 can mirror P1's moves across center/axes, P2 wins. If P1 can make a central move leaving symmetric halves, P1 wins.
        \item \textbf{Invariants \& Parity}: Track $I(S) \bmod 2$ (sum, components, inversions). Control parity to force terminal state on opponent.
        \item \textbf{Pairing Strategy}: Partition board into disjoint matched pairs $(a_i, b_i)$. When P1 takes $a_i$, P2 immediately takes $b_i$.
        \item \textbf{Small-case Pattern Finding}: Simulate for $N \le 20$, compute SG values $g(n)$, find periodicity $g(n) = g(n \bmod T)$ or bit conditions.
    \end{itemize}
    \item \textbf{Sprague-Grundy Theorem (Independent Subgames)}:
    \begin{itemize}
        \item $g(G) = \bigoplus g(G_i)$, where $g(u) = \text{mex}(\{g(v) \mid u \to v\})$. Win $\iff g(G) \ne 0$.
        \item \textbf{Tree Hackenbush (Colon Principle)}: Branch node $g(u) = \bigoplus_{v \in \text{children}} (g(v) + 1)$.
    \end{itemize}
    \item \textbf{Classic Nim Variations}:
    \begin{itemize}
        \item \textbf{Standard Nim}: XOR sum $S = \bigoplus a_i$. Win $\iff S \ne 0$.
        \item \textbf{Bounded Subtraction Nim} (take $1..k$): $g(a) = a \bmod (k + 1) \implies$ Win $\iff \bigoplus (a_i \bmod (k+1)) \ne 0$.
        \item \textbf{Moore's $\text{Nim}_k$} (take from $\le k$ piles): Lose $\iff \forall \text{ bit position } b, \sum_i \text{bit}_b(a_i) \equiv 0 \pmod{k + 1}$.
        \item \textbf{Staircase Nim}: Equivalent to standard Nim on odd-indexed stairs ($1, 3, 5, \dots$).
        \item \textbf{Mis\`ere Nim} (last move LOSES):
        \begin{itemize}
            \item All piles $\le 1$: Win $\iff$ count of size-1 piles is EVEN ($S = 0$).
            \item At least one pile $> 1$: Identical to standard Nim (Win $\iff S \ne 0$).
        \end{itemize}
    \end{itemize}
\end{itemize}

\Algorithm{Probability \& Expected Value}
{Linearity of expectation, Indicator variables, Coupon collector, Inequalities}
{}
{}{}

\begin{itemize}
    \item \textbf{Linearity of Expectation}: $E[X_1 + \dots + X_n] = E[X_1] + \dots + E[X_n]$ ($\forall X_i$).
    \item \textbf{Indicator Variables}: $E[X] = \sum P(A_i)$.
    \item \textbf{Geometric Distribution}: $E[X] = 1/p$, $\text{Var}(X) = (1-p)/p^2$.
    \item \textbf{Coupon Collector}: $E[T] = n \sum_{i=1}^n \frac{1}{i} = n H_n \approx n(\ln n + \gamma)$ ($\gamma \approx 0.5772$).
    \item \textbf{Markov}: $P(X \ge a) \le \frac{E[X]}{a}$; \textbf{Chebyshev}: $P(|X - \mu| \ge k\sigma) \le \frac{1}{k^2}$.
\end{itemize}
