// 1. Iterate all submasks of a given mask:
for sub = mask; sub > 0; sub = (sub - 1) & mask:
    process(sub)
process(0)

// 2. Gosper's Hack: Iterate all subsets of size K from N elements
function gospers_hack(K, N):
    mask = (1 << K) - 1
    while mask < (1 << N):
        process(mask)
        c = mask & -mask, r = mask + c
        mask = (((r ^ mask) >> 2) / c) | r
