// Complexity: O(E * sqrt(V))
// Maximum Bipartite Matching on Left set L [1..n] and Right set R [1..m]

int pair_u[MAXN], pair_v[MAXM], dist[MAXN]

function bfs_hopcroft():
    queue Q
    for u = 1 to n:
        if pair_u[u] == 0: dist[u] = 0; Q.push(u)
        else: dist[u] = INF
    dist[0] = INF
    while not Q.empty():
        u = Q.pop()
        if dist[u] < dist[0]:
            for v in adj[u]:
                if dist[pair_v[v]] == INF:
                    dist[pair_v[v]] = dist[u] + 1
                    Q.push(pair_v[v])
    return dist[0] != INF

function dfs_hopcroft(u):
    if u != 0:
        for v in adj[u]:
            if dist[pair_v[v]] == dist[u] + 1 and dfs_hopcroft(pair_v[v]):
                pair_v[v] = u; pair_u[u] = v
                return True
        dist[u] = INF
        return False
    return True

function max_matching():
    fill(pair_u, 0); fill(pair_v, 0); matching = 0
    while bfs_hopcroft():
        for u = 1 to n:
            if pair_u[u] == 0 and dfs_hopcroft(u):
                matching += 1
    return matching
