// Complexity: O(N log N) or O(N log^2 N) for SA, O(N) for Kasai LCP
// SA[i] = start index of i-th lexicographically smallest suffix; LCP[i] = lcp(S[SA[i]..], S[SA[i-1]..])

function build_suffix_array(string S):
    S = S + "$" // sentinel smallest char
    n = length(S)
    SA[0..n-1] = [0..n-1], rank[0..n-1] = ASCII of S
    
    k = 1
    while k < n:
        // Sort SA by key: (rank[i], rank[(i + k) % n])
        sort(SA, key = (i) -> (rank[i], rank[(i + k) % n]))
        tmp_rank[SA[0]] = 0
        for i = 1 to n - 1:
            prev = SA[i-1], cur = SA[i]
            same = (rank[cur] == rank[prev] and rank[(cur + k) % n] == rank[(prev + k) % n])
            tmp_rank[cur] = tmp_rank[prev] + (0 if same else 1)
        rank = tmp_rank
        if rank[SA[n-1]] == n - 1: break
        k = k * 2
    return SA

function kasai_lcp(string S, SA):
    n = length(S), pos[0..n-1], lcp[0..n-1]
    for i = 0 to n - 1: pos[SA[i]] = i
    k = 0
    for i = 0 to n - 1:
        if pos[i] == 0: continue
        j = SA[pos[i] - 1]
        while i + k < n and j + k < n and S[i + k] == S[j + k]: k = k + 1
        lcp[pos[i]] = k
        if k > 0: k = k - 1
    return lcp
