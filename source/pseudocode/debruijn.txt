// k >= 1, n >= 1; symbols 0..k-1.
// Emits k^n symbols to callback emit; returns nothing.
function de_bruijn(k, n, emit):
    if k == 1:
        emit(0)
        return
    a = array[0..n] filled with 0

    function generate(t, p): // next position, candidate period
        if t > n:
            if n % p == 0:
                for i = 1 to p: emit(a[i])
            return

        a[t] = a[t - p]
        generate(t + 1, p)
        for symbol = a[t - p] + 1 to k - 1:
            a[t] = symbol
            generate(t + 1, t)

    generate(1, 1)
// Example: k=2, n=3 -> 00010111 (cyclic).
