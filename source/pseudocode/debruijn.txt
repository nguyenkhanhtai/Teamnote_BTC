// Complexity: O(K^N) time and space
// Constructs cyclic string of length K^N containing all substrings of length N over alphabet [0..K-1]

function de_bruijn(K, N):
    if K == 1: return "0" * N
    visited_edges = set()
    sequence = []
    
    function dfs(node_str):
        for c in 0 to K - 1:
            edge = node_str + str(c)
            if edge not in visited_edges:
                visited_edges.insert(edge)
                dfs(edge[1..N-1])
                sequence.append(c)
                
    start_node = "0" * (N - 1)
    dfs(start_node)
    // Repeat prefix of length N - 1 at the end to make it cyclic
    return reverse(sequence) + start_node
