// Complexity: O(log m)
// Computes \sum_{i=0}^{n-1} \lfloor (a * i + b) / m \rfloor (Like-Euclid)
// Recognition: Summing floor divisions of linear functions over ranges

function floor_sum(n, m, a, b):
    ans = 0
    if a >= m:
        ans += (n - 1) * n / 2 * (a / m)
        a %= m
    if b >= m:
        ans += n * (b / m)
        b %= m
    y_max = (a * n + b) / m
    x_max = y_max * m - b
    if y_max == 0: return ans
    // Rotate coordinate system: sum over y from 0 to y_max - 1
    ans += (n - (x_max + a - 1) / a) * y_max
    ans += floor_sum(y_max, a, m, (a - x_max % a) % a)
    return ans
