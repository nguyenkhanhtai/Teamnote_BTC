// Complexity: O(N^3) time, O(N^2) space
// Minimum weight perfect matching on 1-indexed N x M bipartite graph (N <= M)

function hungarian(cost[1..N][1..M]):
    u[0..N] = 0, v[0..M] = 0, p[0..M] = 0, way[0..M] = 0
    for i = 1 to N:
        p[0] = i, j0 = 0
        minv[0..M] = INF, used[0..M] = False
        repeat:
            used[j0] = True
            i0 = p[j0], delta = INF, j1 = 0
            for j = 1 to M:
                if not used[j]:
                    cur = cost[i0][j] - u[i0] - v[j]
                    if cur < minv[j]: minv[j] = cur; way[j] = j0
                    if minv[j] < delta: delta = minv[j]; j1 = j
            for j = 0 to M:
                if used[j]: u[p[j]] += delta; v[j] -= delta
                else: minv[j] -= delta
            j0 = j1
        until p[j0] == 0
        repeat:
            j1 = way[j0]; p[j0] = p[j1]; j0 = j1
        until j0 == 0
    // Result: matched left node of right node j is p[j], min cost is -v[0]
    return -v[0]
