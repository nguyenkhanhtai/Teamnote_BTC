// Complexity: O(k) time where k is distance + cycle length, O(1) space
// Floyd's Tortoise and Hare for functional graph (out-degree of each node is 1: f(x))

function find_cycle(x0, f):
    tortoise = f(x0), hare = f(f(x0))
    while tortoise != hare:
        tortoise = f(tortoise)
        hare = f(f(hare))
        
    // Find entry point of cycle (mu)
    mu = 0, tortoise = x0
    while tortoise != hare:
        tortoise = f(tortoise)
        hare = f(hare)
        mu += 1
        
    // Find cycle length (lambda)
    lam = 1, hare = f(tortoise)
    while tortoise != hare:
        hare = f(hare)
        lam += 1
        
    return (entry_node = tortoise, start_dist = mu, cycle_len = lam)
