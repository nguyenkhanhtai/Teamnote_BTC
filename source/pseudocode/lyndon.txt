// Complexity: O(N) time, O(1) space
// Duval's algorithm: factorizes S into Lyndon words s1 >= s2 >= ... >= sk (each si is strictly smaller than all its proper suffixes)

function duval_lyndon_factorization(string S):
    n = length(S), i = 0, factors = []
    while i < n:
        j = i + 1, k = i
        while j < n and S[k] <= S[j]:
            if S[k] < S[j]: k = i
            else: k = k + 1 // periodic continuation
            j = j + 1
        while i <= k:
            factors.append(S[i .. i + j - k - 1])
            i = i + j - k
    return factors
