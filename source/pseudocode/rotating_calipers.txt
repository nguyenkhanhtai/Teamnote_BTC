// CH: CCW convex hull, no duplicate/collinear boundary vertices.
// Returns antipodal index pairs; duplicates may occur. O(n).
function antipodal_pairs(CH):
    n = len(CH)
    if n < 2: return []
    if n == 2: return [(0, 1)]

    function area2(i, k):
        edge = CH[(i + 1) % n] - CH[i]
        return cross(edge, CH[k] - CH[i])

    pairs = []
    j = 1
    for i = 0 to n - 1:
        ni = (i + 1) % n
        while area2(i, (j + 1) % n) > area2(i, j):
            j = (j + 1) % n

        pairs.append((i, j))
        pairs.append((ni, j))
        nj = (j + 1) % n
        if area2(i, nj) == area2(i, j):
            pairs.append((i, nj))
            pairs.append((ni, nj))
    return pairs
