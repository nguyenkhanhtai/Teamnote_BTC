\subsection{Biến đổi bài toán (Transformations)}
\noindent\textbf{Ghi chú gốc}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parsep}{0pt}
    \item \textcolor{red}{\textbf{Khi bí}}: Đọc lại đề! Xét trường hợp nhỏ ($N\le6$) để tìm quy luật/bất biến; suy ra độ phức tạp ($N\le20\to2^N,500\to N^3,5000\to N^2,2\cdot10^5\to N\log N$).

    \item \textbf{Tối ưu $\to$ quyết định (Binary Search)}: Nếu điều kiện $P(X)$ đơn điệu, binary search $X$ qua kiểm tra khả thi.

    \item \textbf{Hoán vị toàn cục $\to$ đổi chỗ cục bộ}: Lập luận trao đổi greedy: đổi $(i,j)\to(j,i)$ cải thiện mục tiêu $\iff f(i,j)<f(j,i)$.

    \item \textbf{Ràng buộc $\to$ Graph / Flow / Cut}: 2-SAT, difference constraints ($x_j-x_i\le w$), Project Selection (min-cut), Hall, Dilworth (DAG path cover $=|V|-\text{MaxMatching}$).

    \item \textbf{Đại số $\to$ hình học/tính lồi}: Đường bao trên/dưới, Convex Hull, Slope Trick, Minkowski sum, CHT.

    \item \textbf{Trung bình có trọng số $\to$ Convex Hull}: Xem $\lambda_i$ là tỉ lệ trộn ($\lambda_i\ge0$, $\sum_i\lambda_i=1$). Khi đó $x=\sum_i\lambda_ix_i$ nằm giữa các đầu vào: hai điểm cho đoạn $AB$, ba điểm cho miền tam giác, nhiều điểm cho convex hull. Mọi điểm trong convex hull đều biểu diễn được như vậy. Trong 1D: $x\in[\min x_i,\max x_i]$. Chuẩn hóa trọng số dương: $\lambda_i=w_i/\sum_jw_j$.

    \item \textbf{Tỉ số tổng có trọng số $\to$ trung bình tỉ số}: Với $w_i\ge0$, $b_i>0$, $\frac{\sum_iw_ia_i}{\sum_iw_ib_i}$ là trung bình có trọng số của $a_i/b_i$, trọng số mới $w_ib_i$. Vì vậy nằm trong $[\min_i(a_i/b_i),\max_i(a_i/b_i)]$. Không trọng số: $w_i=1$.

    \item \textbf{Trộn trước $\to$ Jensen}: Với $f$ lồi (hình cái bát), tính sau khi trộn rẻ hơn: $f(\sum_i\lambda_ix_i)\le\sum_i\lambda_if(x_i)$. Đọc là $f(\text{trung bình})\le\text{trung bình của }f$.

    \item \textbf{Đoạn toàn cục $\to$ Divide \& Conquer}: Chia $[l,r]\to[l,m]$ và $[m+1,r]$, cộng đóng góp chéo (CDQ D\&C, D\&C DP).

    \item \textbf{Xóa offline $\to$ thêm (đảo thời gian)}: Đảo thứ tự truy vấn $Q\to1$ để đổi xóa thành thêm DSU. Game DP: đi ngược từ trạng thái kết thúc.

    \item \textbf{Chia miền tìm kiếm (Meet-in-Middle)}: Chia $N=N/2+N/2$, giảm $2^N\to2^{N/2}$ bằng hash table/two pointers.

    \item \textbf{Đúng $k$ nhóm $\to$ hệ số phạt Lagrange}: Tối ưu $cost+\lambda m$, $m$ là số nhóm. Binary search $\lambda$ khi $m$ đơn điệu; cần tính lồi và xử lý hòa để khôi phục đúng $k$. Bài \emph{đếm} đúng $k$ tập dùng Binomial Inversion/PIE từ số cách thuộc ít nhất $k$ tập.

    \item \textbf{Tổng toàn cục $\to$ đóng góp phần tử}: Đổi góc nhìn: cộng đóng góp mỗi phần tử/cạnh/cặp $(u,v)$ trên mọi trạng thái hợp lệ (Fubini).

    \item \textbf{Chia theo ngưỡng ($\sqrt N$ Heavy/Light)}: Nhóm $\le B$: duyệt trực tiếp; $>B$: tối đa $N/B$ nhóm, cache riêng. Cân bằng chi phí theo $B$; khác kỹ thuật Mo.

    \item \textbf{Nâng cấp cuối $\to$ chặn trạng thái}: Bỏ nâng cấp cuối, lập lại kế hoạch còn lại; nếu chi phí tăng $\le$ giá nâng cấp thì bỏ được. Với $L\le N$, $k>\lceil\sqrt N\rceil$: $\lceil L/(k-1)\rceil-\lceil L/k\rceil\le1$. Giá nâng cấp $\ge1$, mỗi điểm dừng giá 1 $\implies$ chặn tài nguyên ở $\lceil\sqrt N\rceil$.

    \item \textbf{Gom nhóm thương (Quotient Grouping)}: $\lfloor N/k\rfloor$ có $O(\sqrt N)$ giá trị khác nhau. Với $1\le l\le N$, nhóm cùng thương kết thúc ở $r=\lfloor N/\lfloor N/l\rfloor\rfloor$; tiếp theo $l=r+1$.

    \item \textbf{Đếm chính xác $\to$ bù/PIE}: Đếm trạng thái hợp lệ qua $\text{tổng}-\text{không hợp lệ}$, Stars \& Bars, PIE.

    \item \textbf{Impartial Game $\to$ Nim Sum}: Chuyển game trên DAG thành XOR các giá trị Sprague-Grundy ($SG(u)=\text{mex}\{SG(v)\}$).

    \item \textbf{Undirected Vertex Geography $\to$ Matching}: Quân bắt đầu tại $s$; hai người luân phiên đi tới hàng xóm chưa thăm, xóa đỉnh vừa rời. Người đầu thắng iff $s$ thuộc mọi maximum matching của $G$, tương đương $\nu(G)>\nu(G-s)$.

    \item \textbf{Tổng quát $\to$ trường hợp đặc biệt}: Giải giới hạn dễ trước ($A_i\in\{0,1\}$, cây, $N\le6$) để tìm quy luật rồi tổng quát hóa.

    \item \textbf{Vét cạn $\to$ Branch \& Bound}: Cắt nhánh DFS bằng cận hợp lệ.

    \item \textbf{Manhattan $\leftrightarrow$ Chebyshev (xoay $45^\circ$)}: $(x,y)\mapsto(x+y,x-y)\implies|x_1-x_2|+|y_1-y_2|=\max(|x'_1-x'_2|,|y'_1-y'_2|)$. Tách hai chiều thành hai trục 1D độc lập.

    \item \textbf{Phân rã max/min và trị tuyệt đối}: $\max(A,B)=\frac{A+B+|A-B|}{2}$, $\min(A,B)=\frac{A+B-|A-B|}{2}$; $|x-a|=\max(x-a,a-x)$. Với một khoảng cách, xét bốn cặp dấu $(\pm x\pm y)$. Để tối thiểu $\sum_i(|x-a_i|+|y-b_i|)$, tách hai tổng 1D rồi chọn median của $a_i$ và $b_i$.

    \item \textbf{Đếm theo ngưỡng/lớp (Fubini)}: $\sum_{i=1}^nA_i=\sum_{v=1}^{\max A}\text{count}(A_i\ge v)$ (cộng giá trị $\longleftrightarrow$ đếm phần tử $\ge v$).

    \item \textbf{Bình phương $\to$ tổng số lẻ}: $x^2=1+3+5+\cdots+(2x-1)=\sum_{i=1}^x(2i-1)$.

    \item \textbf{Dãy chữ số 1 $\to$ tổng cấp số nhân}: $\underbrace{11\ldots1}_{i\text{ chữ số}}=1+10+\cdots+10^{i-1}=\frac{10^i-1}{9}$.

    \item \textbf{Dãy tăng/giảm $\to$ biến không âm}: Bỏ ràng buộc thứ tự bằng chênh lệch kề. Nếu $A_1\le\cdots\le A_n$, đặt $D_i=A_i-A_{i-1}\ge0$; nếu $A_1\ge\cdots\ge A_n$, đặt $D_i=A_{i-1}-A_i\ge0$ ($i\ge2$). Thứ tự nghiêm ngặt cần $D_i\ge1$. Khôi phục $A_i=A_1+\sum_{j=2}^iD_j$ (tăng), hoặc $A_i=A_1-\sum_{j=2}^iD_j$ (giảm).

    \item \textbf{Range Add $\to$ mảng hiệu/Prefix Sum}: $A_i=\sum_{j=1}^iD_j$ ($D_i=A_i-A_{i-1}$); cộng đoạn $[l,r]\to$ hai cập nhật điểm trên $D$. Tổng tĩnh $\to S_r-S_{l-1}$.

    \item \textbf{Cửa sổ cố định $\to$ lớp modulo}: XOR hai phép flip cửa sổ dài $x$ bắt đầu tại $i,i+1$: phần giao triệt tiêu, chỉ đảo $i,i+x$ $\to$ cùng lớp modulo $x$. Mỗi cửa sổ chứa đúng một vị trí mỗi lớp; đặt $P_r=\bigoplus_{i\equiv r\pmod x}s_i$. Flip đảo mọi $P_r$, nên $P_r\oplus P_0$ ($1\le r<x$) bất biến và đủ phân loại trạng thái; từ toàn 0 đạt được $s$ iff mọi $P_r$ bằng nhau. Cue: triệt tiêu hai cửa sổ kề $\to$ hai đầu cách $x$ $\to$ modulo. Cần phép triệt tiêu hợp lệ (XOR, hoặc cộng với phép nghịch đảo); không suy rộng sang gán/max.

    \item \textbf{Thu nhỏ điều kiện: toàn cục $\to$ cục bộ}: Tìm điều kiện trên các phần tử kề/cửa sổ nhỏ tương đương điều kiện cả dãy. Sau sắp xếp, $\min_{i\ne j}|A_i-A_j|=\min_i(A_{i+1}-A_i)$; $A_j-A_i\ge(j-i)D\iff A'_{i+1}\ge A'_i$ ($A'_i=A_i-iD$). Cấp số cộng $\iff$ mọi bộ ba liên tiếp thỏa $A_i+A_{i+2}=2A_{i+1}$; áp dụng ngang/dọc biến hình chữ nhật cấp số cộng thành vùng cửa sổ $3\times3$ hợp lệ (LFAC2 2011A). Đoạn dài $\ge2$ có majority ($>50\%$) luôn chứa witness dài 2 ($x,x$) hoặc 3 ($x,y,x$).

    \item \textbf{Tích $\to$ tổng}: $ab=\binom{a+b}{2}-\binom a2-\binom b2$; $\prod A_i\to\sum\log A_i$ (hoặc log xác suất).

    \item \textbf{Đẳng thức bitwise}: $A+B=(A\oplus B)+2(A\text{ AND }B)=(A\text{ OR }B)+(A\text{ AND }B)$; $A\oplus B=(A\text{ OR }B)-(A\text{ AND }B)$. Lật bit hai số kề: $\sum_{i=1}^n\text{popcount}((i-1)\oplus i)=2n-\text{popcount}(n)$.

    \item \textbf{Kỹ thuật ngẫu nhiên (Monte Carlo / Las Vegas)}: Xem \S\ref{subsec:randomized}. Nghiệm mật độ $p$: Monte Carlo lặp $k$ lần, lỗi $\le(1-p)^k$; Las Vegas kiểm chứng trong kỳ vọng $O(T/p)$. Fingerprinting ($\bigoplus r_v,\sum r_v$) kiểm tra tập/multiset; Color Coding tách đồ thị con; Schwartz-Zippel kiểm tra tồn tại đại số (Tutte).

    \item \textbf{Dirichlet/Pigeonhole (4-Sum $a_x+a_y=a_z+a_w$)}: Các giá trị nguyên dương phân biệt: Có tối đa $2\max A$ giá trị tổng. Nếu $\binom N2>2\max A$, pigeonhole bảo đảm hai cặp trùng tổng. Hash tổng $\to$ cặp chỉ số trong $O(N^2)$; tính phân biệt của các giá trị bảo đảm bốn chỉ số khác nhau.

    \item \textbf{Boyer-Moore Majority Vote}: Tìm phần tử majority ($>n/2$ lần) trong $O(n)$ thời gian, $O(1)$ bộ nhớ bằng một bộ đếm.

    \item \textbf{Không bao nhau $\to$ Antichain (Sperner)}: Chọn các tập con/bitmask sao cho không tập nào chứa tập khác: tối đa $\binom n{\lfloor n/2\rfloor}$. \textbf{Construction}: lấy tất cả tập có đúng $\lfloor n/2\rfloor$ phần tử (cùng popcount nên không bao nhau). Đạt cận khi được chọn từ toàn bộ tập con; có ràng buộc thêm phải kiểm tra lại. Ví dụ: $N$ là tích $n$ prime khác nhau, chọn các ước không chia nhau $\to$ lấy tích của đúng $\lfloor n/2\rfloor$ prime.
\end{itemize}

\noindent\textbf{Invariant: nhận diện và kiểm tra}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parsep}{0pt}
    \item \textbf{Cách tìm}: Đổi biểu diễn để thao tác thành swap/triệt tiêu. Khác bất biến $\to$ không thể; giống nhau cần chứng minh đủ bằng construction/dạng chuẩn.
    \item \textbf{Tổng / tô màu / modulo}: Chuyển lượng $\to\sum a_i$; đối xứng $\to$ tổng xen kẽ/theo màu. Tìm $w$ sao cho $w\cdot\Delta a=0$ (hoặc $0\pmod m$) với mọi thao tác. XOR cùng lượng hai lần triệt tiêu.
    \item \textbf{Hiệu kề / prefix $\to$ swap}: $a_i'=a_{i-1}+a_{i+1}-a_i$ swap hai hiệu kề. XOR $a_i$ vào hai hàng xóm swap hai prefix XOR $P_{i-1},P_i$. Mọi swap kề được phép $\to$ so multiset; giữ biên $a_1$ (hiệu), $P_n$ (prefix XOR).
    \item \textbf{Hoán đổi $\to$ nhóm vị trí / cặp đối xứng}: Swap từng cặp vị trí: cạnh là swap được phép, độc lập trạng thái $\to$ so multiset mỗi component. Swap cách $k\to$ modulo $k$; nhiều bước nhảy thử gcd, kiểm tra biên. Swap nguyên khối prefix/suffix cùng độ dài $l$, $2l\le n$, giữ thứ tự trong khối $\to$ các cặp đối xứng $(a_i,a_{n+1-i})$ chỉ đổi chỗ/đảo hướng: so multiset các cặp $(\min,\max)$, giữ phần tử giữa nếu $n$ lẻ. Điều kiện đủ khi mọi $1\le l\le\lfloor n/2\rfloor$ được phép (đảo thứ tự $l$ cặp đầu và đảo hướng từng cặp); chỉ cho vài $l$ thì chưa chắc đủ.
    \item \textbf{Parity}: Tìm lượng có tính chẵn/lẻ không đổi: tổng, số bit 1, số vật theo màu, số nghịch thế. Kiểm tra mỗi thao tác làm lượng đó đổi một số chẵn. Với hoán vị phân biệt: parity $=(n-\#\text{cycles})\bmod2$; $k$-cycle tương đương $k-1$ swap, nên $k$ lẻ giữ parity. Mọi 3-cycle sinh đúng hoán vị chẵn; phần tử trùng cho phép đổi parity gán nhãn đích.
    \item \textbf{Cùng lớp dư $\to$ GCD các hiệu}: Đặt $d_i=a_{i+1}-a_i$, $g=\gcd_{i<n}|d_i|$. Khi đó $g=\gcd_{i>1}|a_i-a_1|=\gcd_{i<j}|a_i-a_j|$ (hiệu cặp là tổng hiệu kề). Với $g>0$, mọi $a_i$ nằm trên lưới $a_1+g\mathbb Z$; $g$ là modulo lớn nhất để tất cả cùng dư. Tịnh tiến cả mảng hoặc đổi thứ tự không đổi $g$. Cue: khoảng cách giữa giá trị/cùng lớp dư $\to$ xét gcd hiệu, không phải gcd giá trị. Thêm $2u-v$ giữ lưới vì $u,v\equiv a_1\pmod g\implies2u-v\equiv a_1\pmod g$ (CF1477A). $g=0$: mọi giá trị bằng nhau, xét riêng.
\end{itemize}

\noindent\textbf{Monovariant: mỗi bước tiến gần hơn đến đích}
\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parsep}{0pt}
    \item \textbf{Cách dùng}: Chọn một số đo $\Phi$ (số lỗi, khoảng cách, độ lệch); chứng minh sau mỗi bước nó giảm. Nếu $\Phi$ là số nguyên không âm và giảm ít nhất 1, quá trình dừng sau tối đa $\Phi$ ban đầu bước. Dừng chưa chắc đạt đích: cần chứng minh chưa đạt đích thì vẫn chọn được bước làm giảm $\Phi$.
    \item \textbf{Sửa thứ tự $\to$ số nghịch thế}: Swap hai phần tử kề $a_i>a_{i+1}$ làm số cặp sai thứ tự giảm đúng 1. Số vị trí sai không chắc giảm khi swap.
    \item \textbf{Di chuyển $\to$ khoảng cách/vị trí}: Mỗi bước đưa vật gần đích hơn $\to$ tổng khoảng cách giảm. Vật chỉ dịch trái $\to$ tổng vị trí của các vật giảm, dù số vật giữ nguyên.
    \item \textbf{San bằng $\to$ tổng bình phương}: Chuyển 1 từ số lớn $u$ sang số nhỏ $v$, $u\ge v+2$: $(u-1)^2+(v+1)^2-u^2-v^2=-2(u-v-1)<0$. Tổng không đổi, nhưng độ lệch giảm.
    \item \textbf{Cố định dần $\to$ số chỗ còn phải sửa}: Mỗi bước kéo dài prefix đúng và không phá nó $\to$ số vị trí ngoài prefix giảm. Nếu số đo chính đôi khi không đổi, thêm số đo phụ: bước nào cũng giảm số chính, hoặc giữ số chính và giảm số phụ (cả hai nguyên không âm).
    \item \textbf{Kiểm tra}: Chỉ ``không tăng'' có thể lặp; số thực giảm như $1,1/2,1/4,\ldots$ cũng có thể vô hạn. Nếu chỉ các bước mình chọn làm $\Phi$ giảm, kết luận chỉ áp dụng cho thuật toán đó.
\end{itemize}


\begin{itemize}
    \setlength{\itemsep}{1pt}
    \setlength{\parsep}{0pt}
    \item \textbf{Chia miền $\to$ giảm một nửa số mũ (LFAC2 2014H, 4 sao)}: Tem RGB ngắn nhất phủ chuỗi có $k$ wildcard; chồng tại wildcard được khác màu. $L\ge n/2$: chỉ cần prefix/suffix tương thích. $L\le n/2$: hai đầu dài $L$ rời nhau, chọn đầu ít wildcard hơn, tối đa $\lfloor k/2\rfloor$; liệt kê $3^{k/2}$ thay vì $3^k$.

    \item \textbf{Bộ ba Pythagoras}: Nguyên thủy: $(a,b,c)=(u^2-v^2,2uv,u^2+v^2)$, $u>v\ge1$, $\gcd(u,v)=1$, khác parity. Sinh mọi bộ bằng lấy bội $k(a,b,c)$; nếu $c\le C$, duyệt $u^2+v^2\le C$, $k\le C/c$.

    \item \textbf{Factorial: nhánh nhỏ và nhánh khó (CRRC K)}: Tìm $a!/(b!c!)=N$, chọn $b\ge c$, đặt $d=a-b$. Nếu $d\ge c$, $N\ge\binom{2c}{c}$ chặn $c$; nhánh $d<c$ phải khai thác chia hết theo prime valuations. Với $N\le10^9$, editorial dùng $d\le20$; cutoff thực nghiệm, chưa chứng minh ở đây.
\end{itemize}
