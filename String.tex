\section{Chuỗi}

\Algorithm{Chiến lược giải bài chuỗi}
{Phân loại bài chuỗi, chu kỳ, LCP, substring phân biệt, palindrome và định lý chính}
{}
{}{}

\begin{itemize}
    \item \textbf{Problem Taxonomy \& Core Approches}:
    \begin{itemize}
        \item \textbf{LCP với toàn chuỗi/tìm một mẫu}: Dùng \textbf{Z-Algorithm} hoặc \textbf{KMP}.
        \item \textbf{Tìm nhiều mẫu/DP trên tập chuỗi}: Dùng \textbf{Aho-Corasick Automaton}.
        \item \textbf{Substring phân biệt}: Suffix Array ($\frac{n(n+1)}{2} - \sum LCP_i$) hoặc SAM ($\sum len[u] - len[link[u]]$).
        \item \textbf{Substring thứ $k$/LCS của nhiều chuỗi}: Dùng \textbf{Suffix Automaton (SAM)} hoặc \textbf{Suffix Array + LCP}.
        \item \textbf{Substring đối xứng (palindrome)}: Dùng \textbf{Manacher} (bán kính tại mỗi vị trí) hoặc \textbf{EERTREE / PAM} (cây palindrome).
        \item \textbf{Phép xoay chuỗi nhỏ nhất theo thứ tự từ điển}: Chạy \textbf{Lyndon Factorization (Duval)} trên $S + S$.
        \item \textbf{So sánh substring $O(1)$/chuỗi 2D}: Dùng \textbf{Rolling Hash} ($2^{61}-1$ số nguyên tố Mersenne).
    \end{itemize}
    \item \textbf{Chu kỳ và tính chất border}:
    \begin{itemize}
        \item Chu kỳ nhỏ nhất của chuỗi dài $n$ là $p = n - \pi[n - 1]$ (nếu $n \bmod p == 0 \implies S = (S[0..p-1])^{n/p}$).
        \item \textbf{Fine-Wilf (bổ đề chu kỳ)}: Nếu $S$ có chu kỳ $p, q$ và $|S| \ge p + q - \gcd(p, q) \implies S$ có chu kỳ $\gcd(p, q)$.
        \item \textbf{Dãy border}: Mọi border chia thành $\le O(\log n)$ cấp số cộng $\implies$ Tối ưu DP border xuống $O(n \log n)$.
    \end{itemize}
    \item \textbf{Cận lý thuyết}:
    \begin{itemize}
        \item Số palindrome phân biệt trong chuỗi dài $n$ luôn $\le n$.
        \item Suffix Automaton của chuỗi dài $n$ có $\le 2n - 1$ trạng thái và $\le 3n - 4$ cạnh chuyển.
    \end{itemize}
\end{itemize}

\Algorithm{Suffix Array}
{Dựng suffix array bằng doubling, tính LCP bằng Kasai}
{$O(n \log n)$ dựng, $O(n)$ LCP}
{compactcpp}{source/cpp/string/suffix_array.cpp}

\begin{itemize}
    \item \textbf{Ứng dụng LCP trong thi đấu}:
    \begin{itemize}
        \item \textbf{LCP của hai suffix bất kỳ}: $\text{LCP}(S[i..], S[j..]) = \min_{k=\min(pos[i], pos[j])+1}^{\max(pos[i], pos[j])} LCP[k]$ qua RMQ/Sparse Table, truy vấn $O(1)$.
        \item \textbf{Substring phân biệt}: Tổng substring phân biệt là $\frac{n(n+1)}{2} - \sum_{i=1}^{n-1} LCP[i]$.
        \item \textbf{Substring thứ $k$ theo thứ tự từ điển}: Suffix $SA[i]$ đóng góp $n - SA[i] - LCP[i]$ chuỗi mới, có độ dài $LCP[i]+1 \dots n - SA[i]$.
        \item \textbf{LCS của hai chuỗi}: Dựng SA trên $S_1 + \# + S_2$; đáp án là $\max LCP[i]$ với $SA[i], SA[i-1]$ thuộc hai chuỗi khác nhau.
        \item \textbf{LCS của $K$ chuỗi}: Dựng SA trên $S_1 + \#_1 + \dots + S_K + \#_K$; two pointers/sliding window $[L, R]$ chứa đủ $K$ chuỗi $\implies \min_{k=L+1}^R LCP[k]$.
        \item \textbf{Khoảng xuất hiện mẫu}: Các suffix có prefix $P$ tạo đoạn liên tiếp $[L, R]$ trong SA; tìm bằng hai binary search trong $O(|P| \log n)$, số lần $= R - L + 1$.
        \item \textbf{Substring dài nhất xuất hiện $\ge K$ lần}: $\max_i (\min_{j=i+1}^{i+K-1} LCP[j])$ bằng monotonic queue/sliding RMQ trong $O(n)$.
    \end{itemize}
\end{itemize}
