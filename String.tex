\begin{itemize}
    \item \textbf{Problem Taxonomy \& Core Approches}:
    \begin{itemize}
        \item \textbf{Longest Common Prefix with Whole String / Single Pattern Match}: Use \textbf{Z-Algorithm} or \textbf{KMP}.
        \item \textbf{Multiple Dictionary Matching / DP on String Sets}: Use \textbf{Aho-Corasick Automaton}.
        \item \textbf{Distinct Substrings}: Suffix Array ($\frac{n(n+1)}{2} - \sum LCP_i$) or SAM ($\sum len[u] - len[link[u]]$).
        \item \textbf{$k$-th Substring / LCS of Multiple Strings}: Use \textbf{Suffix Automaton (SAM)} or \textbf{Suffix Array + LCP}.
        \item \textbf{Palindromic Substrings}: Use \textbf{Manacher} (radius centered at every index) or \textbf{EERTREE / PAM} (palindromic tree).
        \item \textbf{Lexicographically Minimal String Rotation}: Run \textbf{Lyndon Factorization (Duval)} on $S + S$.
        \item \textbf{Fast Substring Equivalence $O(1)$ / 2D Strings}: Use \textbf{Rolling Hash} ($2^{61}-1$ Mersenne prime).
    \end{itemize}
    \item \textbf{Periodicity \& Border Properties}:
    \begin{itemize}
        \item Minimal period of length-$n$ string is $p = n - \pi[n - 1]$ (if $n \bmod p == 0 \implies S = (S[0..p-1])^{n/p}$).
        \item \textbf{Fine-Wilf (Periodicity Lemma)}: If $S$ has periods $p, q$ and $|S| \ge p + q - \gcd(p, q) \implies S$ has period $\gcd(p, q)$.
        \item \textbf{Border Series}: All borders of a string partition into $\le O(\log n)$ arithmetic progressions $\implies$ Optimize border DP to $O(n \log n)$.
    \end{itemize}
    \item \textbf{Theoretical Bounds}:
    \begin{itemize}
        \item Distinct palindromic substrings in length-$n$ string is always $\le n$.
        \item Suffix Automaton of length-$n$ string has $\le 2n - 1$ states and $\le 3n - 4$ transition edges.
    \end{itemize}
\end{itemize}

\begin{itemize}
    \item \textbf{LCP Applications \& Core CP Recipes}:
    \begin{itemize}
        \item \textbf{Arbitrary Suffix LCP}: $\text{LCP}(S[i..], S[j..]) = \min_{k=\min(pos[i], pos[j])+1}^{\max(pos[i], pos[j])} LCP[k]$ via RMQ / Sparse Table in $O(1)$ query.
        \item \textbf{Distinct Substrings}: Total distinct substrings is $\frac{n(n+1)}{2} - \sum_{i=1}^{n-1} LCP[i]$.
        \item \textbf{$k$-th Lexicographical Substring}: Suffix $SA[i]$ contributes $n - SA[i] - LCP[i]$ new strings of lengths $LCP[i]+1 \dots n - SA[i]$.
        \item \textbf{LCS of 2 Strings}: Build SA on $S_1 + \# + S_2$; ans is $\max LCP[i]$ where $SA[i], SA[i-1]$ belong to different strings.
        \item \textbf{LCS of $K$ Strings}: Build SA on $S_1 + \#_1 + \dots + S_K + \#_K$; two pointers / sliding window $[L, R]$ covering all $K$ strings $\implies \min_{k=L+1}^R LCP[k]$.
        \item \textbf{Pattern Occurrences Range}: Suffixes with prefix $P$ form continuous range $[L, R]$ in SA; find via 2 binary searches in $O(|P| \log n)$, count $= R - L + 1$.
        \item \textbf{Longest Substring Occurring $\ge K$ Times}: $\max_i (\min_{j=i+1}^{i+K-1} LCP[j])$ via monotonic queue / sliding RMQ in $O(n)$.
    \end{itemize}
\end{itemize}
