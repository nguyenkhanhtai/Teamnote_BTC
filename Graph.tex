\subsection{Đếm thứ tự topo trên cây/rừng}
Cạnh hướng cha $\to$ con: số thứ tự topo là $\displaystyle \frac{n!}{\prod_v sz(v)}$, với $sz(v)$ là kích thước subtree kể cả $v$. Ghép các thứ tự con bằng multinomial; đảo toàn bộ cạnh vẫn cùng số cách. Không áp dụng cho DAG tổng quát.

\Algorithm{MinCut - MaxFlow}
{Dinic: số đường đi/đường đi không chung đỉnh lớn nhất từ source đến sink.
\begin{itemize}
    \item \textbf{Tô hai màu lưới}: Chia ô theo $(r + c) \bmod 2 \implies$ đồ thị hai phía. Dùng lát domino, đặt quân không tấn công nhau, bắn tia laser trên lưới.
    \item \textbf{Tách đỉnh (Vertex Splitting) ($u_{in} \to u_{out}$)}: Giới hạn capacity đỉnh hoặc buộc đi qua đỉnh (capacity $c_u$, chi phí $w_u$).
    \item \textbf{Max-Weight Closure (Project Selection)}:
    \begin{itemize}
        \item Quy tắc: chọn $u$ thì phải chọn $v$ (cạnh có hướng $u \to v$ với $\text{cap} = \infty$).
        \item Trọng số dương $W_u > 0$: thêm cạnh $S \to u$ với $\text{cap} = W_u$.
        \item Trọng số âm $W_v < 0$: thêm cạnh $v \to T$ với $\text{cap} = -W_v$.
        \item $\text{Max Profit} = \sum_{W_u > 0} W_u - \text{MinCut}(S, T)$.
    \end{itemize}
    \item \textbf{Phân nhóm với hai lựa chọn}: Gán $i$ vào $S$ (chi phí $A_i$) hoặc $T$ ($B_i$). Nếu $i \in S, j \in T$, phạt $C_{i,j}$. Thêm $S \to i$ ($B_i$), $i \to T$ ($A_i$), $i \to j$ ($C_{i,j}$). Tổng chi phí $=$ min-cut.
    \item \textbf{Circulation có cận dưới/trên $[L_e, R_e]$}: Thay cạnh $e=(u,v)$ bằng capacity $R_e-L_e$. Đặt $d(u)=\sum_{in}L-\sum_{out}L$. Thêm super-source $S^*$, super-sink $T^*$. Với $d(u)>0$, thêm $S^*\to u$ capacity $d(u)$; với $d(u)<0$, thêm $u\to T^*$ capacity $-d(u)$. Khả thi $\iff$ mọi cạnh từ $S^*$ bão hòa.
\end{itemize}
}
{$\mathcal{O}(EV^2)$}
{compactcpp}{source/cpp/graph/max_flow_dinic.cpp}

\Algorithm{Bipartite Matching}
{Hopcroft-Karp: vertex cover, path cover không chung đỉnh/tổng quát, antichain (Dilworth).
\begin{itemize}
    \item \textbf{Định lý K\H{o}nig}: $\text{Max Matching} = \text{Min Vertex Cover}$.
    \item \textbf{Max Independent Set}: $|V| - \text{Min Vertex Cover} = |V| - \text{Max Matching}$.
    \item \textbf{Min Edge Cover} (không có đỉnh cô lập): $|V| - \text{Max Matching}$.
    \item \textbf{Path cover nhỏ nhất trên DAG}:
    \begin{itemize}
        \item \textit{Không chung đỉnh}: Tách $u \to u_{out}, u_{in}$; dựng matching hai phía $u_{out} \to v_{in}$ cho cạnh $u \to v$. Kết quả: $|V| - \text{Max Matching}$.
        \item \textit{Tổng quát (được chung đỉnh)}: Tính transitive closure (Floyd-Warshall/reachability), rồi giải path cover không chung đỉnh.
    \end{itemize}
    \item \textbf{Định lý Dilworth}: Trong poset hữu hạn (reachability trên DAG), kích thước \textbf{maximum antichain} (tập lớn nhất gồm các đỉnh không đến được nhau) bằng \textbf{minimum path cover}.
    \item \textbf{Erd\H{o}s--Gallai}: Dãy số nguyên $n-1\ge d_1\ge\cdots\ge d_n\ge0$ là dãy bậc đồ thị đơn vô hướng iff $\sum d_i$ chẵn và $\sum_{i=1}^{k}d_i\le k(k-1)+\sum_{i=k+1}^{n}\min(d_i,k)$ với mọi $1\le k\le n$. Dùng khi chỉ biết bậc và cần kiểm tra tồn tại đồ thị.
    \item \textbf{Gale--Ryser}: Hai dãy nguyên không âm $a_1\ge\cdots\ge a_n$, $b_1,\dots,b_m$ là bậc hai phía của đồ thị đơn iff $\sum a_i=\sum b_j$ và $\sum_{i=1}^{k}a_i\le\sum_{j=1}^{m}\min(b_j,k)$ với mọi $1\le k\le n$. Dùng cho ma trận $0/1$ có tổng hàng/cột cho trước.
    \item \textbf{Định lý Hall}: Đồ thị hai phía $(X,Y)$ có matching phủ hết $X \iff \forall S \subseteq X, |N(S)| \ge |S|$.
    \item \textbf{Kích thước maximum matching qua Hall}: $|X| - \max_{S \subseteq X} (|S| - |N(S)|)$.
\end{itemize}
}
{$\mathcal{O}(E\sqrt{V})$}
{compactcpp}{source/cpp/graph/hopcroft_karp.cpp}

\Algorithm{General Matching}
{Edmonds Blossom: maximum matching trên đồ thị tổng quát.
\begin{itemize}
    \item \textbf{Tutte Matrix \& Rabin-Vazirani ($O(V^3)$)}: Với $(u,v)\in E$ ($u<v$), đặt $T_{u, v} = r \in [1, p-1]$, $T_{v, u} = -r \pmod p$. Kích thước $= \frac{1}{2}\text{rank}(T)$ (Gauss mod $p$).
    \begin{itemize}
        \item \textit{Khôi phục Rabin-Vazirani}: Tính $A=T^{-1}\pmod p$ (trên cơ sở $2K\times2K$). Cạnh $(u,v)$ thuộc một matching $\iff A_{u,v}\ne0$. Chọn cặp đó, ghép lại rồi cập nhật $A$ trong $O(V^2)$ bằng Schur complement: $A'_{x, y} = A_{x, y} + \frac{A_{x, u}A_{y, v} - A_{x, v}A_{y, u}}{A_{u, v}} \pmod p$. Lặp cho $K$ cặp $\implies$ tổng $O(V^3)$.
    \end{itemize}
    \item \textbf{Quy về max $b$-matching} ($\deg(i) \le a_i$): Đỉnh $i \to a_i$ slot ($a_i \leftarrow \min(a_i, \deg(i))$). Cạnh $e=(u,v) \to L_e, R_e$, nối $L_e - R_e$, $L_e - \text{slot}(u)$, $R_e - \text{slot}(v)$. Bỏ $e$: match $L_e-R_e$ ($+1$). Chọn $e$: match $L_e, R_e$ với slot ($+2$, lợi thêm $+1$). Đồ thị phụ $|V'| \le 4M \implies$ Blossom.
\end{itemize}
}
{$\mathcal{O}(V^3)$}
{compactcpp}{source/cpp/graph/general_matching.cpp}

\Algorithm{2-SAT}
{Giải 2-SAT bằng đồ thị kéo theo (implication graph) và SCC.
\begin{itemize}
    \item \textbf{Ít nhất một biến đúng ($X \lor Y$)}: Thêm các cạnh $\neg X \to Y$ và $\neg Y \to X$.
    \item \textbf{Nếu $X$ thì $Y$ ($X \implies Y$)}: Thêm các cạnh $X \to Y$ và $\neg Y \to \neg X$.
    \item \textbf{Đúng một biến đúng ($X \text{ XOR } Y = 1$)}: $(X \lor Y) \land (\neg X \lor \neg Y)$. Các cạnh: $X \leftrightarrow \neg Y$ và $\neg X \leftrightarrow Y$.
    \item \textbf{$X$ và $Y$ phải bằng nhau ($X \text{ XOR } Y = 0$)}: $(X \lor \neg Y) \land (\neg X \lor Y)$. Các cạnh: $X \leftrightarrow Y$ và $\neg X \leftrightarrow \neg Y$.
    \item \textbf{Buộc $X$ đúng/sai}: Thêm cạnh $\neg X \to X$ (buộc đúng), hoặc $X \to \neg X$ (buộc sai).
    \item \textbf{Nhiều nhất một biến trong $\{X_1, X_2, \dots, X_k\}$ đúng}: Thêm biến prefix $P_i = P_{i-1} \lor X_i$. Các cạnh: $X_i \to P_i$, $P_{i-1} \to P_i$, và $X_i \to \neg P_{i-1}$ (nếu $X_i$ đúng thì mọi biến trước phải sai).
\end{itemize}
}
{$\mathcal{O}(n + m)$}
{compactcpp}{source/cpp/graph/two_sat.cpp}

\Algorithm{Eulerian Path}
{Hierholzer: tìm đường đi/chu trình Euler.
\begin{itemize}
    \item \textbf{Nối từ/chuỗi (De Bruijn/domino)}: Nối từ sao cho đuôi từ trước khớp đầu từ sau. Đỉnh là \textit{trạng thái} (prefix/suffix dài $L-1$ hoặc nửa domino); cạnh có hướng là \textit{đối tượng}. Đường đi Euler đi qua mỗi đối tượng đúng một lần.
    \item \textbf{Định hướng cạnh vô hướng}: Định hướng cạnh để cân bằng bậc vào/ra (ví dụ $|in-out|\le1$). Thêm cạnh giả (hoặc nối đỉnh bậc lẻ tới đỉnh giả) để mọi bậc chẵn; tìm chu trình Euler rồi định hướng theo lượt duyệt.
    \item \textbf{Phân rã chu trình không chung cạnh}: Phân rã thành chu trình không chung cạnh. Khả thi $\iff$ đồ thị liên thông và mọi đỉnh bậc chẵn. Tìm chu trình Euler rồi tách các chu trình.
    \item \textbf{Duyệt lưới bằng đồ thị hàng-cột}: Bài ma trận di chuyển luân phiên hàng/cột: dựng đồ thị hai phía gồm hàng và cột. Ô $(r,c)$ là cạnh nối hàng $r$ với cột $c$. Đường đi Euler duyệt các ô theo hàng-cột-hàng.
\end{itemize}
}
{$\mathcal{O}(m)$}
{compactcpp}{source/cpp/graph/euler_path.cpp}
