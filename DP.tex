\Algorithm{Divide-and-Conquer DP và bất đẳng thức tứ giác}
{\textbf{Bất đẳng thức tứ giác (Monge)}: $\forall a \le b \le c \le d \implies C(a, c) + C(b, d) \le C(a, d) + C(b, c)$ ($\text{Cross} \le \text{Nest}$).
\begin{itemize}
  \item \textbf{Kiểm tra cục bộ cần và đủ}: Chỉ cần kiểm tra các ô $2 \times 2$ kề nhau (tổng rút gọn):
  $$ C(i, j+1) - C(i, j) \ge C(i+1, j+1) - C(i+1, j) \quad (\forall i < j) $$
  (Tương đương, độ tăng chi phí $\Delta_j(i) = C(i, j+1) - C(i, j)$ không tăng theo $i$; trong trường hợp liên tục $\frac{\partial^2 C}{\partial x \partial y} \le 0$).
  \item \textbf{Các hàm chi phí thường thỏa QI}:
  \begin{itemize}
    \item $C(l, r) = f(S_r - S_{l-1})$ với $f$ lồi ($f'' \ge 0$), ví dụ $(S_r - S_{l-1})^2$, $(S_r - S_{l-1})^k$ ($k \ge 1$).
    \item $C(l, r) = \sum_{k=l}^r w_k = W_r - W_{l-1}$ ($w_k \ge 0$, dấu bằng trong Knuth Optimal BST).
    \item $C(l, r) = \text{inv}(l, r)$ (số nghịch thế trong đoạn $[l, r]$, hoặc tính chất trên cặp chéo).
    \item $C(l, r) = |x_r - x_l|^p$ ($p \ge 1$ với $x$ đã sắp xếp), hoặc phương sai/tổng bình phương chênh lệch.
  \end{itemize}
  \item \textbf{Đơn điệu}: Tính chất Monge $\implies opt(i_1) \le opt(i_2)$ với $i_1 \le i_2$ (DnC: $O(k n \log n)$) và $opt[i][j-1] \le opt[i][j] \le opt[i+1][j]$ (Knuth: $O(n^2)$).
\end{itemize}
}
{$\mathcal{O}(k n \log n)$ với DnC / $\mathcal{O}(n^2)$ với Knuth}
{compactcpp}{source/cpp/dp/dnc_dp.cpp}

\Algorithm{Alien DP}
{DP trên hàm lồi/lõm (WQS Binary Search):
\begin{itemize}
  \item Chi phí lồi/lõm $\to$ Chọn greedy.
  \item DP lồi/lõm $\to$ Binary search hệ số phạt $\lambda$.
\end{itemize}
}
{$\mathcal{O}(n \log (\text{max\_val}))$}
{compactcpp}{source/cpp/dp/alien_dp.cpp}

\subsection{Bitmask DP}
\begin{itemize}
  \item \textbf{Matching}: Tập cặp không có thứ tự $\to$ cố định một đầu, chọn đầu còn lại. $O(n2^n)$ thời gian, $O(2^n)$ bộ nhớ.
  \item \textbf{Nén trạng thái}: $i=\operatorname{popcount}(mask)$ $\to$ $dp[i][mask]\to dp[mask]$.
\end{itemize}

\Algorithm{Expected Value Trick}
{Chọn tối ưu $k$ trong $n$ phần tử đã xáo trộn: bước $i$ dùng heuristic dải ngẫu nhiên (có thể bỏ lỡ tối ưu), chỉ xét trạng thái trong $\left[\frac{i \cdot k}{n} - C\sqrt{n}, \frac{i \cdot k}{n} + C\sqrt{n}\right]$.}
{$\mathcal{O}(n \sqrt{n})$}
