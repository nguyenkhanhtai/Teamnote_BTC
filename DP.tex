\Algorithm{Divide-And-Conquer DP \& Quadrangle Inequality}
{\textbf{Quadrangle Inequality (Monge)}: $\forall a \le b \le c \le d \implies C(a, c) + C(b, d) \le C(a, d) + C(b, c)$ ($\text{Cross} \le \text{Nest}$).
\begin{itemize}
  \item \textbf{Sufficient \& Necessary Local Test}: Checking adjacent $2 \times 2$ cells is sufficient (telescoping sum):
  $$ C(i, j+1) - C(i, j) \ge C(i+1, j+1) - C(i+1, j) \quad (\forall i < j) $$
  (Equivalently, cost increase $\Delta_j(i) = C(i, j+1) - C(i, j)$ is non-increasing in $i$; or continuous $\frac{\partial^2 C}{\partial x \partial y} \le 0$).
  \item \textbf{Common Cost Functions Satisfying QI}:
  \begin{itemize}
    \item $C(l, r) = f(S_r - S_{l-1})$ for any convex $f$ ($f'' \ge 0$), e.g. $(S_r - S_{l-1})^2$, $(S_r - S_{l-1})^k$ ($k \ge 1$).
    \item $C(l, r) = \sum_{k=l}^r w_k = W_r - W_{l-1}$ ($w_k \ge 0$, exact equality $=$ in Knuth Optimal BST).
    \item $C(l, r) = \text{inv}(l, r)$ (inversion count in subarray $[l, r]$, or any cross-pair property).
    \item $C(l, r) = |x_r - x_l|^p$ ($p \ge 1$ on sorted $x$), or variance / sum of squared differences.
  \end{itemize}
  \item \textbf{Monotonicity}: Monge property $\implies opt(i_1) \le opt(i_2)$ for $i_1 \le i_2$ (DnC: $O(k n \log n)$) and $opt[i][j-1] \le opt[i][j] \le opt[i+1][j]$ (Knuth: $O(n^2)$).
\end{itemize}
}
{$\mathcal{O}(k n \log n)$ for DnC / $\mathcal{O}(n^2)$ for Knuth}
{}{source/pseudocode/dnc_dp.txt}

\subsection{Bitmask DP}
\begin{itemize}
  \item \textbf{Matching}: Unordered set of pairs $\to$ fix one endpoint, choose its partner. $O(n2^n)$ time, $O(2^n)$ space.
  \item \textbf{State Compression}: $i=\operatorname{popcount}(mask)$ $\to$ $dp[i][mask]\to dp[mask]$.
\end{itemize}
