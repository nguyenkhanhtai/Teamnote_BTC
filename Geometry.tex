\Algorithm{Geometry Formulas}
{Planar theorems, circles, triangle centers, circle inversion \& 3D primitives}
{}
{}{}

\begin{itemize}
    \item \textbf{Pick's Theorem}: $S = I + \frac{B}{2} - 1 \implies I = S - \frac{B}{2} + 1$, where $B = \sum \gcd(|x_{i+1} - x_i|, |y_{i+1} - y_i|)$.
    \item \textbf{Euler's Planar Formula}: $V - E + F = 1 + C$ ($V - E + F = 2$ for connected planar graphs).
    \item \textbf{Ptolemy's Theorem}: Cyclic quad $ABCD \iff AC \cdot BD = AB \cdot CD + BC \cdot AD$.
    \item \textbf{Stewart's Theorem}: $b^2m + c^2n = a(d^2 + mn)$ (cevian $d$ divides side $a$ into $m, n$).
    \item \textbf{Bretschneider's Formula} (General quadrilateral area):
    $$ S = \sqrt{(p-a)(p-b)(p-c)(p-d) - abcd \cos^2\left(\tfrac{A+C}{2}\right)} $$
    \textit{(Cyclic quad $\implies \cos(\frac{A+C}{2}) = 0$, reduces to Brahmagupta's formula)}.
    \item \textbf{Descartes' Theorem} (4 mutually tangent circles):
    $$ \left(\sum_{i=1}^4 k_i\right)^2 = 2 \sum_{i=1}^4 k_i^2 \quad \left(k_i = \pm \frac{1}{R_i}\right) $$
    \item \textbf{Circle Inversion}: Center $O$, radius $R$. $P \mapsto P'$ satisfies $OP \cdot OP' = R^2$. Maps lines not through $O \longleftrightarrow$ circles through $O$; preserves angles and tangencies.
    \item \textbf{Circle-Circle Intersection}: Let $d = |O_1 - O_2|$. Intersects when $|R_1 - R_2| \le d \le R_1 + R_2$, angle $\alpha = \text{acos}\left(\frac{R_1^2 + d^2 - R_2^2}{2 R_1 d}\right)$.
    \item \textbf{Apollonius Circle}: Locus $\frac{PA}{PB} = k$ ($k \ne 1$) is a circle with diameter connecting the internal and external division points of $AB$ with ratio $k$.
    \item \textbf{Circumcenter $O$ \& Orthocenter $H$}: Let $d_B = x_B^2+y_B^2-x_A^2-y_A^2$, $d_C = x_C^2+y_C^2-x_A^2-y_A^2$:
    $$x_O = \frac{d_B (y_C - y_A) - d_C (y_B - y_A)}{D}$$
    $$y_O = \frac{(x_B - x_A) d_C - (x_C - x_A) d_B}{D}$$
    $$D = 2 \left[ (x_B - x_A)(y_C - y_A) - (x_C - x_A)(y_B - y_A) \right]$$
    $$H = A + B + C - 2O$$

    \item \textbf{3D Primitives}:
    \begin{itemize}
        \item 3D Cross Product: $\vec{A} \times \vec{B} = (A_y B_z - A_z B_y, \; A_z B_x - A_x B_z, \; A_x B_y - A_y B_x)$.
        \item Tetrahedron Volume: $V = \frac{1}{6} \left| ((B - A) \times (C - A)) \cdot (D - A) \right|$.
        \item Plane through $P_0$ with normal $\vec{n}$: $\vec{n} \cdot (P - P_0) = 0$; Point-plane distance: $\frac{|\vec{n} \cdot (P - P_0)|}{|\vec{n}|}$.
    \end{itemize}
\end{itemize}
