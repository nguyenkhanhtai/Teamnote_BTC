\Algorithm{Công thức hình học}
{Định lý phẳng, đường tròn, tâm tam giác, phép nghịch đảo và phép toán 3D}
{}
{}{}

\begin{itemize}
    \item \textbf{Biến đổi tuyến tính $\leftrightarrow$ ma trận}: Mỗi ánh xạ tuyến tính $T:\mathbb{R}^n\to\mathbb{R}^m$ có $A=[T(e_1)\;\cdots\;T(e_n)]$ sao cho $T(x)=Ax$. Các dạng 2D thường dùng:
    \begin{itemize}
        \item Quay góc $\theta$: $\begin{pmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{pmatrix}$; co giãn: $\begin{pmatrix}s_x&0\\0&s_y\end{pmatrix}$.
        \item Đối xứng qua $Ox$, $Oy$, $y=x$: $\begin{pmatrix}1&0\\0&-1\end{pmatrix}$, $\begin{pmatrix}-1&0\\0&1\end{pmatrix}$, $\begin{pmatrix}0&1\\1&0\end{pmatrix}$.
        \item Shear (trượt): $\begin{pmatrix}1&k\\0&1\end{pmatrix}$ hoặc $\begin{pmatrix}1&0\\k&1\end{pmatrix}$; chiếu lên vector đơn vị $u$: $P=uu^T$.
        \item Tịnh tiến là affine, không tuyến tính: $x'=Ax+b$. Trong tọa độ thuần nhất, $\begin{pmatrix}x'\\y'\\1\end{pmatrix}=\begin{pmatrix}a&b&t_x\\c&d&t_y\\0&0&1\end{pmatrix}\begin{pmatrix}x\\y\\1\end{pmatrix}$.
        \item Hợp thành $T_2\circ T_1$ ứng với $A_2A_1$; lặp một biến đổi $k$ lần ứng với $A^k$.
    \end{itemize}
    \item \textbf{Định lý Pick}: $S = I + \frac{B}{2} - 1 \implies I = S - \frac{B}{2} + 1$, với $B = \sum \gcd(|x_{i+1} - x_i|, |y_{i+1} - y_i|)$.
    \item \textbf{Công thức Euler cho đồ thị phẳng}: $V - E + F = 1 + C$ ($V - E + F = 2$ với đồ thị phẳng liên thông).
    \item \textbf{Định lý Ptolemy}: Tứ giác nội tiếp $ABCD \iff AC \cdot BD = AB \cdot CD + BC \cdot AD$.
    \item \textbf{Định lý Stewart}: $b^2m + c^2n = a(d^2 + mn)$ (cevian $d$ chia cạnh $a$ thành $m, n$).
    \item \textbf{Công thức Bretschneider} (Diện tích tứ giác bất kỳ):
    $$ S = \sqrt{(p-a)(p-b)(p-c)(p-d) - abcd \cos^2\left(\tfrac{A+C}{2}\right)} $$
    \textit{(Tứ giác nội tiếp $\implies \cos(\frac{A+C}{2}) = 0$, trở thành công thức Brahmagupta)}.
    \item \textbf{Định lý Descartes} (bốn đường tròn tiếp xúc từng đôi):
    $$ \left(\sum_{i=1}^4 k_i\right)^2 = 2 \sum_{i=1}^4 k_i^2 \quad \left(k_i = \pm \frac{1}{R_i}\right) $$
    \item \textbf{Nghịch đảo đường tròn (Circle Inversion)}: Tâm $O$, bán kính $R$. $P \mapsto P'$ thỏa $OP \cdot OP' = R^2$. Đổi đường thẳng không qua $O \longleftrightarrow$ đường tròn qua $O$; bảo toàn góc và tiếp xúc.
    \item \textbf{Giao hai đường tròn}: Đặt $d = |O_1 - O_2|$. Có giao khi $|R_1 - R_2| \le d \le R_1 + R_2$, góc $\alpha = \text{acos}\left(\frac{R_1^2 + d^2 - R_2^2}{2 R_1 d}\right)$.
    \item \textbf{Đường tròn Apollonius}: Quỹ tích $\frac{PA}{PB} = k$ ($k \ne 1$) là đường tròn có đường kính nối hai điểm chia trong và ngoài $AB$ theo tỉ số $k$.
    \item \textbf{Tâm ngoại tiếp $O$ và trực tâm $H$}: Đặt $d_B = x_B^2+y_B^2-x_A^2-y_A^2$, $d_C = x_C^2+y_C^2-x_A^2-y_A^2$:
    $$x_O = \frac{d_B (y_C - y_A) - d_C (y_B - y_A)}{D}$$
    $$y_O = \frac{(x_B - x_A) d_C - (x_C - x_A) d_B}{D}$$
    $$D = 2 \left[ (x_B - x_A)(y_C - y_A) - (x_C - x_A)(y_B - y_A) \right]$$
    $$H = A + B + C - 2O$$

    \item \textbf{Phép toán 3D}:
    \begin{itemize}
        \item Cross product 3D: $\vec{A} \times \vec{B} = (A_y B_z - A_z B_y, \; A_z B_x - A_x B_z, \; A_x B_y - A_y B_x)$.
        \item Thể tích tứ diện: $V = \frac{1}{6} \left| ((B - A) \times (C - A)) \cdot (D - A) \right|$.
        \item Mặt phẳng qua $P_0$ có pháp tuyến $\vec{n}$: $\vec{n} \cdot (P - P_0) = 0$; Khoảng cách điểm đến mặt phẳng: $\frac{|\vec{n} \cdot (P - P_0)|}{|\vec{n}|}$.
    \end{itemize}
\end{itemize}
