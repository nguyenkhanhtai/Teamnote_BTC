\Algorithm{Lazy Segment Tree (Implementation Tips)}
{Lazy propagation discipline, tag compositions (Affine, Range Set), and $O(\log n)$ Tree Walk}
{}
{}{}

\begin{itemize}
    \item \textbf{Push/Pull Discipline}: Call \texttt{push(id, l, r)} at the start of \texttt{update} / \texttt{query}; call \texttt{pull(id)} (\texttt{st[id] = merge(st[2*id], st[2*id+1])}) after updating children.
    \item \textbf{Multiple Lazy Tags}:
    \begin{itemize}
        \item \textbf{Affine ($x \mapsto a \cdot x + b$)}: Composing new $(c, d)$ on $(a, b) \implies c(ax+b)+d = (ca)x + (cb+d)$. \\ Tag: \texttt{laz\_a = laz\_a * c; laz\_b = laz\_b * c + d;} \\ Node: \texttt{val = val * c + (r - l + 1) * d;}
        \item \textbf{Range Set vs Range Add}: Range Set ($x \leftarrow v$) overrides previous Range Add $\implies$ reset \texttt{add = 0}.
    \end{itemize}
    \item \textbf{Tree Walk ($O(\log n)$)}: If left child satisfies condition ($\max \ge X$ or sum $\ge S$), push lazy and recurse left; else adjust target and recurse right.
    \item \textbf{Bounds}: Allocate $4N$ for 1-based indexing (\texttt{id << 1} and \texttt{id << 1 | 1}).
\end{itemize}

\Algorithm{Cartesian Tree}
{Binary tree combining heap-order on values and BST-order on indices}
{$\mathcal{O}(n)$ build via Monotonic Stack}
{}{}

\begin{itemize}
    \item \textbf{Properties}: Min-Heap on values ($val[\text{parent}] \le val[\text{child}]$) + In-order yields original array indices ($1..n$). Root is global minimum. Subtree of $u$ corresponds to the maximal subarray $[L_u, R_u]$ where $A[u]$ is the minimum ($size(u) = R_u - L_u + 1$).
    \item \textbf{$\mathcal{O}(n)$ Stack Build}: Maintain the right spine in a stack. For $i = 1..n$: pop $u$ with $A[u] > A[i]$; last popped becomes $left[i] = u$; remaining top gets $right[\text{top}] = i$. Push $i$.
    \item \textbf{Key Applications}:
    \begin{itemize}
        \item \textbf{RMQ $\iff$ LCA}: Range minimum $\min_{l \le k \le r} A[k] = A[\text{LCA}(l, r)]$.
        \item \textbf{Max Rectangle in Histogram}: $\max_u (A[u] \cdot size(u))$.
        \item \textbf{D\&C to Tree DP}: Transforms array D\&C splitting at minimum into subtree DP.
    \end{itemize}
\end{itemize}
